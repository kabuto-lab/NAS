  ```rust
  //! `xtask capability-coverage` · ENTITY §14, §18.3.
  //!
  //! Greps `crates/presentation/src/**/*.rs` for handler-shaped
  //! functions and requires each to either:
  //!   (a) carry a `caps.require(<cap-name>)` marker in its doc
  //!       comment OR in the immediately-preceding line range
  //!       (max 30 lines above the signature), OR
  //!   (b) carry a `// no_capability_required: <reason>` opt-out.
  //!
  //! Heuristic detection — NOT a full AST pass. Trade-offs:
  //!   + Zero rustc dep; runs without nightly / proc-macro
  //!   + Fast (single regex pass per file)
  //!   - Doesn't follow into nested modules' const-fn-defined
  //!     functions; we don't have those today
  //!   - Doesn't understand `#[handler]`-attributed fns specifically
  //!     (axum has no such attr); we infer via signature shape
  //!
  //! "Handler-shaped" =
  //!   `pub async fn <name>(...) -> ...` where the file path matches
  //!   `crates/presentation/src/api/**/*.rs`.

  use std::path::{Path, PathBuf};

  use eyre::{eyre, Result, WrapErr};
  use regex::Regex;
  use walkdir::WalkDir;

  const PRESENTATION_API_PREFIX: &str = "crates/presentation/src/api";
  const MARKER_LOOKBACK_LINES: usize = 30;

  pub fn run() -> Result<()> {
      let root = workspace_root()?;
      let api_dir = root.join(PRESENTATION_API_PREFIX);
      if !api_dir.is_dir() {
          println!(
              "xtask capability-coverage: no handler directory at {} — \
               skipping (presentation crate has no api/ yet)",
              api_dir.display()
          );
          return Ok(());
      }

      let handler_sig = Regex::new(
          r"^\s*pub\s+async\s+fn\s+(\w+)\s*\("
      ).unwrap();
      let marker_re = Regex::new(
          r"caps\.require\s*\(\s*([a-z0-9._]+)\s*\)"
      ).unwrap();
      let opt_out_re = Regex::new(
          r"no_capability_required\s*[:=]\s*\S+"
      ).unwrap();

      let mut violations: Vec<String> = Vec::new();
      let mut handler_count = 0usize;

      for entry in WalkDir::new(&api_dir).into_iter().filter_map(|e| e.ok()) {
          let path = entry.path();
          if !is_rust_source(path) { continue; }
          let body = std::fs::read_to_string(path)
              .wrap_err_with(|| format!("read {}", path.display()))?;
          let lines: Vec<&str> = body.lines().collect();

          for (idx, line) in lines.iter().enumerate() {
              let Some(caps) = handler_sig.captures(line) else { continue; };
              handler_count += 1;
              let handler_name = caps.get(1).map_or("?", |m| m.as_str());

              // Look back N lines (inclusive of current — but the
              // signature line itself rarely carries the marker).
              let lo = idx.saturating_sub(MARKER_LOOKBACK_LINES);
              let window: String = lines[lo..=idx].join("\n");

              if opt_out_re.is_match(&window) {
                  continue;
              }
              if marker_re.is_match(&window) {
                  continue;
              }
              violations.push(format!(
                  "{file}:{line}: handler `{handler_name}` lacks \
                   `caps.require(<cap>)` marker (or `no_capability_required: <reason>`)",
                  file = path.strip_prefix(&root).unwrap_or(path).display(),
                  line = idx + 1,
                  handler_name = handler_name,
              ));
          }
      }

      if violations.is_empty() {
          println!(
              "xtask capability-coverage: ok ({} handler(s) scanned in {})",
              handler_count, PRESENTATION_API_PREFIX
          );
          Ok(())
      } else {
          for v in &violations { eprintln!("ERROR: {v}  [ENTITY §14]"); }
          Err(eyre!(
              "xtask capability-coverage: {} handler(s) missing capability gate",
              violations.len()
          ))
      }
  }

  fn workspace_root() -> Result<PathBuf> {
      let manifest = std::env::var("CARGO_MANIFEST_DIR")
          .wrap_err("CARGO_MANIFEST_DIR not set")?;
      Path::new(&manifest).parent()
          .ok_or_else(|| eyre!("xtask manifest has no parent"))
          .map(Path::to_path_buf)
   }

  fn is_rust_source(p: &Path) -> bool {
      p.is_file() && p.extension().and_then(|e| e.to_str()) == Some("rs")
  }

  #[cfg(test)]
  #[allow(clippy::expect_used, clippy::unwrap_used, clippy::panic)]
  mod tests {
      use regex::Regex;

      #[test]
      fn handler_signature_matches_pub_async_fn() {
          let re = Regex::new(r"^\s*pub\s+async\s+fn\s+(\w+)\s*\(").unwrap();
          assert!(re.is_match("pub async fn get_page_by_slug("));
          assert!(re.is_match("    pub async fn list_pages("));
          assert!(!re.is_match("async fn private_helper("));   // not pub
          assert!(!re.is_match("pub fn sync_handler("));        // not async
      }

      #[test]
      fn marker_regex_captures_cap_name() {
          let re = Regex::new(r"caps\.require\s*\(\s*([a-z0-9._]+)\s*\)").unwrap();
          let caps = re.captures("/// caps.require(cms.page.read)").unwrap();
          assert_eq!(caps.get(1).unwrap().as_str(), "cms.page.read");
      }

      #[test]
      fn opt_out_regex_matches_no_capability_required() {
          let re = Regex::new(r"no_capability_required\s*[:=]\s*\S+").unwrap();
          assert!(re.is_match("// no_capability_required: health probe"));
          assert!(re.is_match("// no_capability_required = csp_nonce_endpoint"));
          assert!(!re.is_match("// nope"));
      }

      // The full file-scan happy path is exercised by P3 (sanity sweep
      // against the real handler from W3 D5).
  }
  ```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Marker on multi-line attribute.** `#[axum::debug_handler]
     pub async fn x(...)` — our regex matches the second line. OK.
  2. **Generics on the signature.** `pub async fn x<T>(...)` — our
     regex doesn't match `<T>` between fn name and `(`. Adjust:
     `pub\s+async\s+fn\s+(\w+)(?:<[^>]*>)?\s*\(` — better safe than
     surprised; include this in the regex from the start.
  3. **Doc comment with `caps.require` example.** A code sample
     `/// ```rust\n/// caps.require(cms.page.read)\n/// ```` inside
     a doc-comment WILL satisfy the gate. Acceptable false-positive
     (the example author intends the requirement).
  4. **Nested module handlers.** `api/admin/mod.rs` with handlers
     inside — `WalkDir` recurses so we catch them. The walkdir
     traversal cost is fine for a single-digit-file directory.
  5. **Heuristic limits.** Method-style handlers (impl blocks
     hosting `pub async fn ...`) aren't matched today. If you ever
     write them, extend the regex or migrate to a syn-backed pass.

PRE-RESOLVED DEFAULTS
═════════════════════════════════════════════════════════════════════

  • Heuristic, not AST. Documented limitation.
  • 30-line look-back window — covers doc comments + immediately
    preceding code; large enough not to miss, small enough not to
    false-positive on unrelated `caps.require` mentions.
  • Opt-out marker `no_capability_required: <reason>` — REQUIRES a
    non-empty reason (`\S+`) to discourage drive-by exemption.
