  ```rust
  //! `GET /api/v1/pages/:slug` — fetch a published page by slug.

  use axum::{
      extract::{Extension, Path, State},
      Json,
  };
  use nas2_application::queries::GetPublishedPageBySlug;
  use nas2_common::AppError;
  use nas2_domain::{Post, PostSlug};
  use nas2_tenant::TenantContext;

  use crate::app_state::AppState;
  use crate::caps::extract_caps_for_today;

  /// Public page lookup. Capability requirement: `cms.page.read`.
  ///
  /// caps.require(cms.page.read)
  ///
  /// (Marker above is mandatory — `xtask capability-coverage` greps
  /// it. Removing it without renaming the extractor pattern is a
  /// CI-breaking change.)
  pub async fn get_page_by_slug(
      State(state): State<AppState>,
      Extension(tctx): Extension<TenantContext>,
      Path(slug_raw): Path<String>,
  ) -> Result<Json<Post>, AppError> {
      let slug = PostSlug::try_new(&slug_raw)?;
      let caps = extract_caps_for_today();
      let uc = GetPublishedPageBySlug::new(state.post_repo.clone());
      let post = uc.execute(tctx.tenant_id, tctx.site_id, &slug, &caps).await?;
      Ok(Json(post))
  }
  ```

  Two senior-architect notes embedded:

  - `state.post_repo.clone()` is an `Arc::clone` (cheap; no heap alloc).
  - The use case is constructed per request — zero state, so this is
    fine; if a use case were to gain state (caches), we'd hold it on
    AppState instead.

P5 · `crates/presentation/tests/get_page_by_slug.rs` (NEW)

  ```rust
  #![allow(clippy::expect_used, clippy::unwrap_used, clippy::panic)]

  use std::{collections::HashMap, sync::Arc};

  use axum::{body::Body, http::{Request, StatusCode, header}};
  use chrono::Utc;
  use mockall::mock;
  use nas2_application::ports::PostRepository;
  use nas2_common::{PostId, SiteId, TenantId};
  use nas2_domain::{Post, PostSlug, PostStatus};
  use nas2_presentation::{AppState, build_router};
  use nas2_tenant::{InMemoryTenantResolver, TenantContext};
  use tower::ServiceExt;

  mock! {
      pub Repo {}
      #[async_trait::async_trait]
      impl PostRepository for Repo {
          async fn find_by_slug(&self, _t: TenantId, _s: SiteId, _slug: &PostSlug)
              -> Result<Option<Post>, nas2_common::AppError>;
          async fn find_by_id(&self, _t: TenantId, _id: PostId)
              -> Result<Option<Post>, nas2_common::AppError>;
          async fn list_published(&self, _t: TenantId, _s: SiteId, _p: u32, _pp: u32)
              -> Result<nas2_common::Page<Post>, nas2_common::AppError>;
          async fn insert(&self, _p: &Post) -> Result<(), nas2_common::AppError>;
          async fn update(&self, _p: &Post) -> Result<(), nas2_common::AppError>;
      }
  }

  fn sample_post(slug: &str, status: PostStatus, tenant: TenantId, site: SiteId) -> Post {
      Post {
          id: PostId::new_v4(),
          site_id: site,
          tenant_id: tenant,
          slug: PostSlug::try_new(slug).unwrap(),
          title: "Hello".into(),
          status,
          published_at: matches!(status, PostStatus::Published).then(Utc::now),
          blocks: Vec::new(),
          custom_type: None,
      }
  }

  fn build_state(host: &str, repo: Arc<MockRepo>) -> AppState {
      let tid = TenantId::new_v4();
      let sid = SiteId::new_v4();
      let mut map = HashMap::new();
      map.insert(host.to_owned(), TenantContext::new(tid, sid));
      let resolver = Arc::new(InMemoryTenantResolver::new(map));
      AppState::new(repo, resolver)
  }

  #[tokio::test]
  async fn returns_200_for_published_page() {
      let host = "example.com";
      let mut repo = MockRepo::new();
      repo.expect_find_by_slug().returning(|t, s, _slug| {
          let p = sample_post("hello", PostStatus::Published, t, s);
          Box::pin(async move { Ok(Some(p)) })
      });
      let app = build_router(build_state(host, Arc::new(repo)));
      let req = Request::builder()
          .uri("/api/v1/pages/hello")
          .header(header::HOST, host)
          .body(Body::empty()).unwrap();
      let resp = app.oneshot(req).await.unwrap();
      assert_eq!(resp.status(), StatusCode::OK);
      let bytes = axum::body::to_bytes(resp.into_body(), 16 * 1024).await.unwrap();
      // Just verify it's valid JSON with a `slug` field — full parse
      // belongs to the next day's 401/404/400 tests.
      let v: serde_json::Value = serde_json::from_slice(&bytes).unwrap();
      assert_eq!(v.get("slug").and_then(|s| s.as_str()), Some("hello"));
  }
  ```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Marker comment line.** `xtask capability-coverage` (W4 D2)
     greps for `caps.require(<cap.name>)` near the handler. Place it
     INSIDE the handler's doc comment OR within ~10 lines before the
     `pub async fn` signature. The W4 D2 regex implementation will
     be tolerant within a window.
  2. **`Path<String>` for slug**. Direct extraction; we `try_new` it
     into `PostSlug` inside the handler so 400 maps via `?`.
  3. **`State<AppState>` + `Extension<TenantContext>` ordering** is
     irrelevant to axum — extractors are matched by type.
  4. **`Box::pin` inside mockall closures** is the recurring pattern
     because the trait methods are async. Don't fight it.
  5. **`axum::body::to_bytes` requires a `usize` cap.** Don't pass
     `usize::MAX` — that defeats the safety. 16 KiB is generous for
     a single Post JSON.
