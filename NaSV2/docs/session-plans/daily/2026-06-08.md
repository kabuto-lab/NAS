        pub struct GetPublishedPageBySlug<R: PostRepository> {
            repo: R,
        }
        impl<R: PostRepository> GetPublishedPageBySlug<R> {
            pub fn new(repo: R) -> Self
            pub async fn execute(
                &self,
                tenant: TenantId,
                site: SiteId,
                slug: &PostSlug,
                caps: &CapabilitySet,
            ) -> Result<Post, AppError> {
                if !caps.contains(Capability::CmsPageRead) {
                    return Err(AppError::Forbidden("cms.page.read required".into()));
                }
                match self.repo.find_by_slug(tenant, site, slug).await? {
                    Some(p) if p.status == PostStatus::Published => Ok(p),
                    Some(_) => Err(AppError::NotFound(format!("post {slug:?} not published"))),
                    None => Err(AppError::NotFound(format!("post {slug:?} missing"))),
                }
            }
        }
