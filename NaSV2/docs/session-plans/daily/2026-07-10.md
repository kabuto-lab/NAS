```rust
//! Use case: paginated list of Published pages on a site, gated by
//! Capability::CmsPageRead.

use nas2_common::{AppError, Page, SiteId, TenantId};
use nas2_domain::{Capability, CapabilitySet, Post};

use crate::ports::PostRepository;

pub struct ListPublishedPages<R> { repo: R }

impl<R: PostRepository> ListPublishedPages<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        site: SiteId,
        page: u32,
        per_page: u32,
        caps: &CapabilitySet,
    ) -> Result<Page<Post>, AppError> {
        if !caps.contains(Capability::CmsPageRead) {
            return Err(AppError::Forbidden(Capability::CmsPageRead.as_str().to_owned()));
        }
        self.repo.list_published(tenant, site, page, per_page).await
    }
}
```

  Add 3 unit tests via MockPostRepository (forbidden, empty page,
  populated page).

P2 · `crates/presentation/src/api/pages.rs::list_pages` handler

```rust
use axum::extract::Query;
use serde::Deserialize;

#[derive(Debug, Deserialize)]
pub struct PageQuery {
    #[serde(default = "default_page")]
    page: u32,
    #[serde(default = "default_per_page")]
    per_page: u32,
}
fn default_page() -> u32 { 1 }
fn default_per_page() -> u32 { 20 }

/// caps.require(cms.page.read)
pub async fn list_pages(
    State(state): State<AppState>,
    Extension(tctx): Extension<TenantContext>,
    Extension(caps): Extension<CapabilitySet>,
    Query(q): Query<PageQuery>,
) -> Result<Json<nas2_common::Page<nas2_domain::Post>>, AppError> {
    use nas2_application::queries::ListPublishedPages;
    let uc = ListPublishedPages::new(state.post_repo.clone());
    let page = uc.execute(tctx.tenant_id, tctx.site_id, q.page, q.per_page, &caps).await?;
    Ok(Json(page))
}
```
