P1 · `BulkAction` enum (closed) + use case

  `crates/application/src/commands/bulk_update_status.rs` (NEW):

```rust
use chrono::Utc;
use nas2_common::{AppError, PostId, SiteId, TenantId};
use nas2_domain::{Capability, CapabilitySet, PostStatus};

use crate::ports::PostRepository;

#[derive(Debug, Clone, Copy, serde::Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum BulkAction {
    Archive,
    SetDraft,
    SetPublished,
    SetScheduled,
}

impl BulkAction {
    pub const fn target_status(self) -> PostStatus {
        match self {
            Self::Archive      => PostStatus::Archived,
            Self::SetDraft     => PostStatus::Draft,
            Self::SetPublished => PostStatus::Published,
            Self::SetScheduled => PostStatus::Scheduled,
        }
    }
    pub const fn required_cap(self) -> Capability {
        match self {
            // Setting/changing publish state requires publish cap;
            // archive is "soft delete" — also publish cap.
            _ => Capability::CmsPagePublish,
        }
    }
}

#[derive(Debug, Clone)]
pub struct BulkResult {
    pub action: BulkAction,
    pub successes: Vec<PostId>,
    pub failures: Vec<(PostId, String)>,
}

pub struct BulkUpdateStatus<R> { repo: R }

impl<R: PostRepository> BulkUpdateStatus<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        site: SiteId,
        action: BulkAction,
        ids: Vec<PostId>,
        caps: &CapabilitySet,
    ) -> Result<BulkResult, AppError> {
        if !caps.contains(action.required_cap()) {
            return Err(AppError::Forbidden(action.required_cap().as_str().into()));
        }
        let target = action.target_status();
        let now = Utc::now();
        let mut successes = Vec::with_capacity(ids.len());
        let mut failures = Vec::new();

        // Per-post transaction — failures DON'T affect others.
        // Sequential rather than join_all: keeps DB load predictable
        // for bulk ops up to 100 posts. Could parallelize if needed.
        for id in ids {
            match self.repo.find_by_id(tenant, id).await {
                Ok(Some(mut post)) => {
                    // Domain FSM gate.
                    match post.transition_to(target, now) {
                        Ok(()) => {
                            match self.repo.update(&post).await {
                                Ok(()) => successes.push(id),
                                Err(e) => failures.push((id, format!("update failed: {e}"))),
                            }
                        }
                        Err(e) => failures.push((id, format!("transition rejected: {e}"))),
                    }
                }
                Ok(None)  => failures.push((id, "not found".into())),
                Err(e)    => failures.push((id, format!("lookup failed: {e}"))),
            }
        }
        Ok(BulkResult { action, successes, failures })
    }
}
```

P2 · Bulk result cache (moka)

  `crates/presentation/src/admin/bulk_cache.rs` (NEW):

```rust
use std::sync::Arc;
use std::time::Duration;

use moka::future::Cache;
use uuid::Uuid;

use nas2_application::commands::bulk_update_status::BulkResult;

#[derive(Clone)]
pub struct BulkResultCache {
    inner: Cache<Uuid, Arc<BulkResult>>,
}

impl BulkResultCache {
    pub fn new() -> Self {
        Self {
            inner: Cache::builder()
                .max_capacity(1000)
                .time_to_live(Duration::from_secs(60))
                .build(),
        }
    }
    pub async fn store(&self, result: BulkResult) -> Uuid {
        let token = Uuid::new_v4();
        self.inner.insert(token, Arc::new(result)).await;
        token
    }
    pub async fn fetch(&self, token: Uuid) -> Option<Arc<BulkResult>> {
        self.inner.get(&token).await
    }
}
```

P3 · Handler implementation (replace W2 D4 placeholder)

  `crates/presentation/src/admin/posts.rs`:

```rust
pub async fn posts_bulk_action(
    _: RequireCaps<CmsPagePublish>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Extension(bulk_cache): Extension<BulkResultCache>,
    Form(form): Form<BulkActionForm>,
) -> Result<Response, AppError> {
    let action: BulkAction = serde_json::from_str(&format!("\"{}\"", form.action))
        .map_err(|_| AppError::Validation("unknown bulk action".into()))?;
    let ids: Vec<PostId> = form.ids.iter()
        .filter_map(|s| uuid::Uuid::parse_str(s).ok())
        .map(PostId::from_uuid).collect();
    if ids.is_empty() {
        return Err(AppError::Validation("no posts selected".into()));
    }
    if ids.len() > 100 {
        return Err(AppError::Validation("max 100 posts per bulk action".into()));
    }
    let cmd = BulkUpdateStatus::new(state.post_repo.clone());
    let result = cmd.execute(tctx.tenant_id, tctx.site_id, action, ids, &caps).await?;
    let token = bulk_cache.store(result).await;
    Ok(Redirect::to(&format!("/admin/posts?bulk_result={}", token.simple())).into_response())
}
```

P5 · `<BulkResultBanner/>` component

```rust
#[component]
pub fn BulkResultBanner(result: Arc<BulkResult>) -> impl IntoView {
    let n_ok = result.successes.len();
    let n_fail = result.failures.len();
    let action_label = match result.action {
        BulkAction::Archive      => "archived",
        BulkAction::SetDraft     => "set to draft",
        BulkAction::SetPublished => "published",
        BulkAction::SetScheduled => "scheduled",
    };
    let kind = if n_fail == 0 { "success" } else if n_ok == 0 { "error" } else { "partial" };
    view! {
        <aside class=format!("banner banner-{kind}") role="status">
            <strong>{format!("Bulk action: {n_ok} {action_label}, {n_fail} failed")}</strong>
            <Show when=move || n_fail > 0 fallback=|| ()>
                <details>
                    <summary>"Show failures"</summary>
                    <ul>
                        {result.failures.iter().map(|(id, reason)| view! {
                            <li><code>{id.to_string()}</code> ": " {reason.clone()}</li>
                        }).collect_view()}
                    </ul>
                </details>
            </Show>
        </aside>
    }
}
```

PERFORMANCE / SECURITY / PITFALLS
═════════════════════════════════════════════════════════════════════

  - Sequential per-post processing: 100 posts × ~5ms/post ≈ 500ms.
    Acceptable for occasional admin bulk ops; not for hot path.
  - Result cache TTL 60s — long enough for user to click redirect,
    short enough to prevent stale-result leak.
  - Cross-tenant ID smuggled in `ids[]`: `find_by_id` runs under
    `with_tenant` (the requester's tenant); RLS returns None for
    other tenants' IDs → treated as "not found". Defence working.
  - Action enum parse via JSON wrap is awkward; consider
    `serde_plain` or a manual `FromStr`. Today's JSON-quote trick
    works but is a hack — log as polish follow-up.
  - Banner `<details>` exposes failure reasons including raw
    AppError messages; ensure those don't leak DB internals
    (M2 W6 D4's `classify_pg_err` masks SQL details — verify).
