P2 · `PgPostRepository::list_admin` extended

  Dynamic WHERE building from PostsFilter — use sqlx::QueryBuilder
  for safe parameterised composition:

```rust
async fn list_admin(
    &self,
    tenant: TenantId,
    site: SiteId,
    page: u32,
    per_page: u32,
    sort: PostSortOrder,
    filter: PostsFilter,  // NEW
) -> Result<Page<Post>, AppError> {
    use sqlx::{QueryBuilder, Postgres};

    let per_page = per_page.clamp(1, 100);
    let page = page.max(1);
    let offset = u64::from(page - 1) * u64::from(per_page);
    let order_clause = sort_to_order_clause(sort);

    with_tenant(&self.pool, tenant, |tx| async move {
        // Author + term filters require JOIN against users / post_terms.
        let needs_term_join = filter.term.is_some();

        let mut qb: QueryBuilder<Postgres> = QueryBuilder::new(
            "SELECT p.id, p.site_id, p.tenant_id, p.slug, p.title, p.status, \
                    p.published_at, p.blocks, p.custom_type, p.author_id, \
                    u.email AS author_email, \
                    COUNT(*) OVER() AS total \
             FROM posts p \
             LEFT JOIN users u ON u.id = p.author_id"
        );
        if needs_term_join {
            qb.push(" INNER JOIN post_terms pt ON pt.post_id = p.id");
        }
        qb.push(" WHERE p.site_id = ");
        qb.push_bind(site.into_uuid());

        if let Some(status) = filter.status {
            qb.push(" AND p.status = ");
            qb.push_bind(status_to_str(status));
        }
        if let Some(author) = filter.author {
            qb.push(" AND p.author_id = ");
            qb.push_bind(author.into_uuid());
        }
        if let Some(term) = filter.term {
            qb.push(" AND pt.term_id = ");
            qb.push_bind(term.into_uuid());
        }

        qb.push(format!(" ORDER BY p.{order_clause} LIMIT "));
        qb.push_bind(i64::from(per_page));
        qb.push(" OFFSET ");
        qb.push_bind(i64::try_from(offset).map_err(|_| AppError::Validation("offset overflow".into()))?);

        let rows: Vec<PostRowWithAuthor> = qb.build_query_as().fetch_all(&mut **tx).await
            .map_err(|e| { tracing::error!(error=%e, "list_admin"); AppError::Internal })?;
        let total = rows.first().map(|r| u64::try_from(r.total).unwrap_or(0)).unwrap_or(0);
        let items: Result<Vec<_>, _> = rows.into_iter().map(|r| r.into_domain_with_author()).collect();
        Ok(Page::new(items?, total, page, per_page))
    }).await
}
```

  `PostRowWithAuthor` is a private FromRow struct that mirrors
  `PostRow` + `author_email: Option<String>` + `total: i64`. The
  conversion to `(Post, author_label)` is one shared row → one
  Post, no N+1.

  Note: PostRepository trait surface today carries Post — author label
  belongs to PostRow in presentation. Today's repo method returns
  `Page<Post>`; presentation handler re-fetches authors? Or do we
  return a wider type?

  Decision: return `Page<Post>` (Post gains `author_id: Option<UserId>`);
  presentation does a SECOND query to resolve `Vec<UserId> →
  HashMap<UserId, email>` for display (batched IN-query, one
  roundtrip, no N+1). This keeps the trait surface clean.

P3 · Author batch resolver in presentation

  In `admin/posts.rs`:

```rust
async fn resolve_authors(
    user_repo: &dyn UserRepository,
    tenant: TenantId,
    ids: &[UserId],
) -> Result<HashMap<UserId, String>, AppError> {
    let unique_ids: HashSet<UserId> = ids.iter().copied().collect();
    let mut map = HashMap::new();
    for id in unique_ids {
        if let Some(u) = user_repo.find_by_id(tenant, id).await? {
            map.insert(id, u.email.as_str().to_owned());
        }
    }
    Ok(map)
}
```

  This is technically N queries (1 per unique user). For an admin
  page showing ≤ 100 posts, unique authors ≈ 5-20 — acceptable
  short-term. **TODO:** Add `UserRepository::find_by_ids(&[UserId])
  → HashMap<UserId, User>` next month for real batch — log as

P4 · `<PostListFilters/>` component

```rust
#[component]
pub fn PostListFilters(
    current: PostsFilter,
    available_authors: Vec<(UserId, String /* email */)>,
    available_terms: Vec<(TermId, String /* label */)>,
) -> impl IntoView {
    let active = current.is_active();
    view! {
        <form method="get" class="post-list-filters">
            <select name="status">
                <option value="" selected=current.status.is_none()>"All statuses"</option>
                {[PostStatus::Draft, PostStatus::Scheduled,
                  PostStatus::Published, PostStatus::Archived].into_iter().map(|s| {
                    let v = status_to_str(s);
                    let sel = current.status == Some(s);
                    view! { <option value=v selected=sel>{v}</option> }
                }).collect_view()}
            </select>
            <select name="author">
                <option value="" selected=current.author.is_none()>"All authors"</option>
                {available_authors.into_iter().map(|(id, email)| {
                    let v = id.to_string();
                    let sel = current.author == Some(id);
                    view! { <option value=v selected=sel>{email}</option> }
                }).collect_view()}
            </select>
            <select name="term">
                <option value="" selected=current.term.is_none()>"All terms"</option>
                {available_terms.into_iter().map(|(id, label)| {
                    let v = id.to_string();
                    let sel = current.term == Some(id);
                    view! { <option value=v selected=sel>{label}</option> }
                }).collect_view()}
            </select>
            <button type="submit" class="btn">"Filter"</button>
            <Show when=move || active fallback=|| ()>
                <a href="/admin/posts" class="clear-link">"Clear filters"</a>
            </Show>
        </form>
    }
}
```
