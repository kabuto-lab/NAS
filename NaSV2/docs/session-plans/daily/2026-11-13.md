    let cookie = /* ... existing cookie code ... */;
    let mut headers = HeaderMap::new();
    headers.insert(header::SET_COOKIE, /* ... */);
    headers.insert(header::LOCATION, HeaderValue::from_str(&redirect_to)
        .map_err(|_| AppError::Internal)?);

    // For form-based POST → 303 SEE_OTHER + redirect (browsers GET the location).
    // For JSON API clients → 200 + body (existing behavior).
    let wants_html = /* sniff Accept */;
    if wants_html {
        Ok((StatusCode::SEE_OTHER, headers).into_response())
    } else {
        let body = LoginResponse { /* existing fields */, redirect: redirect_to };
        Ok((StatusCode::OK, headers, Json(body)).into_response())
    }

/// Open-redirect guard: only return path if it's a relative path
/// (starts with single `/`, NOT `//` or `http`).
fn sanitize_next(raw: &str) -> Option<String> {
    let decoded = match url::form_urlencoded::parse(raw.as_bytes()).next() {
        Some((k, _)) if k.as_ref().starts_with('/') => k.to_string(),
        _ => return None,
    };
    if decoded.starts_with('/') && !decoded.starts_with("//") &&
       !decoded.contains("://") && !decoded.contains('\\') {
        Some(decoded)
    } else {
        None
    }
}

P2 · `crates/presentation/static/admin/session.js` (NEW, ~50 lines)

```javascript
// Session-expired detector for admin XHR.
// Listens for fetch responses on the same page; if a 401/403 with
// JSON {"code":"expired_token"} or {"code":"replay_or_logout"} is
// returned, shows a modal and redirects to /admin/login?next=<path>.
//
// No framework dep. Uses fetch monkey-patch.

(function () {
  const originalFetch = window.fetch;
  window.fetch = async function (...args) {
    const resp = await originalFetch.apply(this, args);
    const url = (args[0] && args[0].url) || args[0] || "";
    if (typeof url === "string" && url.startsWith("/api/")) {
      if (resp.status === 401 || resp.status === 403) {
        try {
          const clone = resp.clone();
          const body = await clone.json();
          if (body.code === "expired_token" || body.code === "replay_or_logout"
              || body.code === "missing_token") {
            showSessionExpiredModal();
          }
        } catch (_) { /* not JSON, ignore */ }
      }
    }
    return resp;
  };

  function showSessionExpiredModal() {
    if (document.getElementById("ax-session-modal")) return;
    const div = document.createElement("div");
    div.id = "ax-session-modal";
    div.setAttribute("role", "alertdialog");
    div.innerHTML = `
      <div class="ax-modal-backdrop"></div>
      <div class="ax-modal-content">
        <h2>Session expired</h2>
        <p>Your session has timed out. Redirecting to login in <span id="ax-modal-countdown">3</span>s…</p>
        <button id="ax-modal-relogin">Log in now</button>
      </div>`;
    document.body.appendChild(div);
    const next = encodeURIComponent(location.pathname + location.search);
    const loginUrl = `/admin/login?next=${next}`;
    document.getElementById("ax-modal-relogin").addEventListener("click", () => {
      location.href = loginUrl;
    });
    let n = 3;
    const t = setInterval(() => {
      n--;
      const el = document.getElementById("ax-modal-countdown");
      if (el) el.textContent = n;
      if (n <= 0) { clearInterval(t); location.href = loginUrl; }
    }, 1000);
  }
})();
```

PERFORMANCE / SECURITY CONSIDERATIONS
═════════════════════════════════════════════════════════════════════

  - Open-redirect is the only real security concern of the day —
    sanitize_next() must REJECT:
      `//evil.com/path`  (protocol-relative)
      `http://evil.com`  (absolute URL)
      `\\evil.com`       (back-slash → some old browsers treat as `/`)
      `/admin/../../etc` (path traversal — Note: NOT explicitly
                          checked because browser URL normalisation
                          handles it; the redirect is still relative
                          so impact is contained to our domain)
  - JS is ~1.5 KB unminified — fine for shipping as-is.
  - Modal blocks the page — UX trade-off vs silent redirect. Modal
    is more visible.
  - `defer` attribute means JS executes after HTML parse — no FOUC,
    no blocking.

═════════════════════════════════════════════════════════════════════
COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`url::form_urlencoded::parse`** parses query-string-style
     pairs; for a single `next=<path>` we read the FIRST key. If `raw`
     contains `=`, the parse loses data. Better: use `urlencoding::
     decode(raw)` — but that adds dep. Fallback: write a tiny
     hand-decoder.
  2. **Multiple Set-Cookie in 1 response** — axum supports via
     `headers.append(header::SET_COOKIE, ...)` (NOT `insert`).
  3. **Login form's `action=` with `?next=`** double-encoded in
     `byte_serialize` then Leptos auto-escapes — verify the
     rendered HTML doesn't double-encode the `%2F` to `%252F`.
     Add a test for this exact case.
  4. **session.js stops working if user disables JS** — fallback
     is "next page they click → guard → server-side redirect to
     login". Acceptable.
  5. **Fetch monkey-patch breaks libraries that hold a reference
     to the original fetch** — Leptos doesn't, htmx doesn't. If
     a library does, swap to a wrapper API.
