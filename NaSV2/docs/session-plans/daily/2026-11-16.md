//! Server-rendered PostList table · M7 W2 D1.
//!
//! NO client-side hydration today — pure SSR. Sort/filter controls
//! land W2 D2-D3 as hydrated islands. The Leptos `view!` here is
//! sync-rendered server-side.

use leptos::prelude::*;
use leptos_router::components::A;
use nas2_common::Page;
use nas2_domain::{Post, PostStatus};

#[derive(Clone, Debug)]
pub struct PostListInitialData {
    pub page: Page<PostRow>,
}

/// Lighter row representation (less JSON over the wire for hydration
/// later). Computed server-side from Post + a future caps check.
#[derive(Clone, Debug)]
pub struct PostRow {
    pub id_str: String,         // display only
    pub slug: String,
    pub title: String,
    pub status: PostStatus,
    pub updated_at: chrono::DateTime<chrono::Utc>,
    pub author_label: String,   // resolved name or email
}

#[component]
pub fn PostList(data: PostListInitialData) -> impl IntoView {
    let total = data.page.total;
    let items = data.page.items;
    let is_empty = items.is_empty();
    view! {
        <section class="post-list">
            <header class="post-list-header">
                <h2>"All posts"</h2>
                <A href="/admin/posts/new" attr:class="btn btn-primary">"New Post"</A>
            </header>
            <Show when=move || is_empty
                  fallback=move || view! { <PostListTable rows=items.clone() total/> }>
                <div class="empty-state">
                    <p>"No posts yet."</p>
                    <A href="/admin/posts/new" attr:class="btn btn-primary">"Create your first post"</A>
                </div>
            </Show>
        </section>
    }
}

#[component]
fn PostListTable(rows: Vec<PostRow>, total: u64) -> impl IntoView {
    view! {
        <table class="post-list-table">
            <thead>
                <tr>
                    <th>"Title"</th>
                    <th>"Status"</th>
                    <th>"Author"</th>
                    <th>"Updated"</th>
                    <th class="actions-col">"Actions"</th>
                </tr>
            </thead>
            <tbody>
                {rows.into_iter().map(|r| {
                    let edit_url = format!("/admin/posts/{}/edit", r.slug);
                    let updated_human = humanize_relative(r.updated_at);
                    view! {
                        <tr>
                            <td><A href=edit_url>{r.title.clone()}</A></td>
                            <td><StatusBadge status=r.status/></td>
                            <td class="muted">{r.author_label}</td>
                            <td class="muted" title=r.updated_at.to_rfc3339()>
                                {updated_human}
                            </td>
                            <td class="actions-col">
                                {/* placeholder dropdown — W4 wires real actions */}
                                <span class="muted">"…"</span>
                            </td>
                        </tr>
                    }
                }).collect_view()}
            </tbody>
        </table>
        <footer class="post-list-footer">
            <span class="muted">{format!("{total} total")}</span>
        </footer>
    }
}

#[component]
fn StatusBadge(status: PostStatus) -> impl IntoView {
    let (class, label) = match status {
        PostStatus::Draft     => ("draft", "Draft"),
        PostStatus::Scheduled => ("scheduled", "Scheduled"),
        PostStatus::Published => ("published", "Published"),
        PostStatus::Archived  => ("archived", "Archived"),
    };
    view! { <span class=format!("badge status-{class}")>{label}</span> }
}

/// Render relative time like "2 hours ago" / "yesterday" / "3 days ago".
fn humanize_relative(t: chrono::DateTime<chrono::Utc>) -> String {
    let now = chrono::Utc::now();
    let diff = now.signed_duration_since(t);
    let s = diff.num_seconds();
    match s {
        s if s < 60         => "just now".to_owned(),
        s if s < 3600       => format!("{} min ago", s / 60),
        s if s < 86_400     => format!("{} hours ago", s / 3600),
        s if s < 86_400 * 2 => "yesterday".to_owned(),
        s if s < 86_400 * 30 => format!("{} days ago", s / 86_400),
        _                    => t.format("%Y-%m-%d").to_string(),
    }
}

use axum::{extract::{Extension, State, Query}, response::{IntoResponse, Html}};
use leptos::prelude::*;
use nas2_application::queries::ListPublishedPages;
use nas2_common::{AppError, Page};
use nas2_domain::{CapabilitySet, Post};
use nas2_tenant::TenantContext;

use crate::admin::context::AdminContext;
use crate::admin::layout::AdminLayout;
use crate::admin::breadcrumb::{BreadcrumbChain, Crumb};
use crate::admin::guard::CmsPageRead;
use crate::admin::post_list::{PostList, PostListInitialData, PostRow};
use crate::app_state::AppState;

#[derive(Debug, serde::Deserialize)]
pub struct PostsListQuery {
    #[serde(default = "default_page")]
    page: u32,
    #[serde(default = "default_per_page")]
    per_page: u32,
}
fn default_page() -> u32 { 1 }
fn default_per_page() -> u32 { 25 }

// caps.require(cms.page.read)
pub async fn posts_list(
    _: RequireCaps<CmsPageRead>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Query(q): Query<PostsListQuery>,
) -> Result<impl IntoResponse, AppError> {
    // Use case from M2 W7 D5; today we just consume it.
    let uc = ListPublishedPages::new(state.post_repo.clone());
    let page = uc.execute(tctx.tenant_id, tctx.site_id, q.page, q.per_page, &caps).await?;

    // Convert Vec<Post> → Vec<PostRow> (lighter shape for SSR).
    // Author resolution: today we use a placeholder; future enhancement
    // joins users for `author_label`.
    let rows: Vec<PostRow> = page.items.into_iter().map(|p| PostRow {
        id_str: p.id.to_string(),
        slug: p.slug.as_str().to_owned(),
        title: p.title,
        status: p.status,
        updated_at: p.published_at.unwrap_or_else(chrono::Utc::now),
        author_label: "—".to_owned(),  // placeholder; real lookup later
    }).collect();
    let data = PostListInitialData {
        page: Page::new(rows, page.total, page.page, page.per_page),
    };

    let ctx = build_admin_ctx(&state, /* user_id, tctx, caps, csrf */);
    let crumbs = BreadcrumbChain::root().push(Crumb::terminal("Posts"));

    let html = leptos::ssr::render_to_string(move || view! {
        <AdminLayout ctx title="Posts" crumbs>
            <PostList data/>
        </AdminLayout>
    });
    Ok(Html(html.into_owned()))
}

use chrono::{Utc, Duration};

fn sample_row(slug: &str, status: PostStatus) -> PostRow {
    PostRow {
        id_str: uuid::Uuid::new_v4().to_string(),

    assert!(html.contains("Create your first post"));
    assert!(!html.contains("<table"));

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`#[component]` requires the function to start with capital
     letter** — `PostList` not `post_list`.
  2. **`.into_any()` for divergent view types** in match arms (e.g.
     match on Option). Today's component doesn't need it.
  3. **Author label "—"** is a string em-dash; can stay literal in
     SSR HTML, no encoding issue. Future enhancement: use `<abbr>`.
  4. **`Page<PostRow>` vs `Page<Post>`** — we convert at the handler
     boundary, NOT in the use case. Use case stays domain-typed.
  5. **`<A>` from leptos_router** vs raw `<a>` — both work for SSR.
     `<A>` enables client-side navigation when hydrated (M7 W2+).
     Today we use it for forward compat.
