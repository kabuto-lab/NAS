        // Axum middleware that resolves Host → TenantId via
        // simple in-memory map (deferred to repo lookup in month 2).
        use axum::{extract::{Request, State}, middleware::Next, response::Response};

        pub async fn resolve_tenant(
            State(resolver): State<TenantResolver>,
            mut req: Request,
            next: Next,
        ) -> Result<Response, AppError> {
            let host = req.headers().get("host")
                .and_then(|v| v.to_str().ok())
                .ok_or_else(|| AppError::Validation("missing host header".into()))?;
            let ctx = resolver.resolve(host)?;
            req.extensions_mut().insert(ctx);
            Ok(next.run(req).await)
        }

        // Простой resolver — реальный repo lookup в month 2.
        #[derive(Clone)]
        pub struct TenantResolver {
            map: std::sync::Arc<std::collections::HashMap<String, TenantContext>>,
        }
        impl TenantResolver {
            pub fn from_map(m: HashMap<String, TenantContext>) -> Self
            pub fn resolve(&self, host: &str) -> Result<TenantContext, AppError>
        }
