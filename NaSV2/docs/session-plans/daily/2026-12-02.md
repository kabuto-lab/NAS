//! Update alt + caption on a Media row.

use nas2_common::{AppError, MediaId, TenantId};
use nas2_domain::{Capability, CapabilitySet};

use crate::ports::MediaRepository;

pub struct UpdateMedia<R> { repo: R }

#[derive(Debug, Clone)]
pub struct UpdateMediaInput {
    pub alt: Option<String>,
    pub caption: Option<String>,
}

impl<R: MediaRepository> UpdateMedia<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        id: MediaId,
        input: UpdateMediaInput,
        caps: &CapabilitySet,
    ) -> Result<(), AppError> {
        if !caps.contains(Capability::MediaUpload) {
            return Err(AppError::Forbidden(Capability::MediaUpload.as_str().into()));
        }
        self.repo.update_metadata(tenant, id, input.alt, input.caption).await
    }
}

async fn update_metadata(
    &self, tenant: TenantId, id: MediaId,
    alt: Option<String>, caption: Option<String>,
) -> Result<(), AppError> {
    with_tenant(&self.pool, tenant, |tx| async move {
        let result = sqlx::query(
            "UPDATE media SET \
             alt = COALESCE($1, alt), \
             caption = COALESCE($2, caption), \
             updated_at = now() \
             WHERE id = $3"
        ).bind(alt).bind(caption).bind(id.into_uuid())
         .execute(&mut **tx).await
         .map_err(|e| { tracing::error!(error=%e, "update media"); AppError::Internal })?;
        if result.rows_affected() == 0 {
            return Err(AppError::NotFound(format!("media {id}")));
        }
        Ok(())
    }).await
}

// caps.require(media.upload)
pub async fn patch_media(
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Path(id_raw): Path<String>,
    Json(body): Json<PatchMediaBody>,
) -> Result<Json<UpdatedMediaResp>, AppError> {
    let id = MediaId::from_uuid(uuid::Uuid::parse_str(&id_raw)
        .map_err(|_| AppError::Validation("invalid id".into()))?);
    let cmd = UpdateMedia::new(state.media_repo.clone());
    cmd.execute(tctx.tenant_id, id, UpdateMediaInput {
        alt: body.alt, caption: body.caption,
    }, &caps).await?;
    Ok(Json(UpdatedMediaResp { id: id.to_string() }))
}

```rust
use leptos::prelude::*;
use nas2_common::MediaId;

#[derive(Clone, Debug)]
pub struct MediaDetail {
    pub id: MediaId,
    pub mime: String,
    pub bytes_size: u64,
    pub width: Option<u32>,
    pub height: Option<u32>,
    pub checksum: String,
    pub original_url: String,
    pub variants: Vec<(String, String, u64)>,  // (kind, url, bytes)
    pub alt: String,
    pub caption: Option<String>,
}

#[component]
pub fn MediaDetailPanel(
    detail: MediaDetail,
    on_close: Callback<()>,
) -> impl IntoView {
    let ctx = expect_context::<crate::admin::context::AdminContext>();
    let alt = RwSignal::new(detail.alt.clone());
    let caption = RwSignal::new(detail.caption.clone().unwrap_or_default());
    let save_status = RwSignal::new("");
    let id = detail.id;
    let api_base = ctx.api_base.clone();
    let csrf = ctx.csrf_token.clone();

    let on_save = move |_| {
        save_status.set("Saving…");
        let alt_v = alt.get_untracked();
        let cap_v = caption.get_untracked();
        let url = format!("{api_base}/media/{id}");
        let csrf = csrf.clone();
        let status = save_status.clone();
        spawn_local(async move {
            let body = serde_json::json!({
                "alt": alt_v, "caption": if cap_v.is_empty() { None } else { Some(cap_v) },
            }).to_string();
            let req = web_sys::RequestInit::new();
            req.set_method("PATCH");
            req.set_body(&body.into());
            let h = web_sys::Headers::new().unwrap();
            let _ = h.set("Content-Type", "application/json");
            let _ = h.set("X-CSRF-Token", &csrf);
            req.set_headers(&h);
            let r = web_sys::Request::new_with_str_and_init(&url, &req).unwrap();
            let p = web_sys::window().unwrap().fetch_with_request(&r);
            if let Ok(resp_val) = wasm_bindgen_futures::JsFuture::from(p).await {
                let resp: web_sys::Response = resp_val.dyn_into().unwrap();
                if resp.ok() { status.set("Saved"); } else { status.set("Save failed"); }
            } else {
                status.set("Network error");
            }
        });
    };

    view! {
        <aside class="media-detail-panel" on:keydown=move |ev: web_sys::KeyboardEvent| {
            if ev.key() == "Escape" { on_close.call(()); }
        }>
            <header>
                <h3>"Media details"</h3>
                <button class="modal-close" on:click=move |_| on_close.call(())>"×"</button>
            </header>
            <div class="detail-preview">
                <img src=detail.original_url alt=move || alt.get()/>
            </div>
            <div class="detail-meta">
                <dl>
                    <dt>"MIME"</dt><dd><code>{detail.mime}</code></dd>
                    <dt>"Size"</dt><dd>{human_bytes(detail.bytes_size)}</dd>
                    {detail.width.zip(detail.height).map(|(w,h)| view! {
                        <>
                            <dt>"Dimensions"</dt>
                            <dd>{format!("{w}×{h}")}</dd>
                        </>
                    })}
                    <dt>"Checksum"</dt><dd><code class="muted">{detail.checksum.chars().take(16).collect::<String>()}…</code></dd>
                    <dt>"Variants"</dt><dd>{detail.variants.len()} " generated"</dd>
                </dl>
            </div>
            <div class="detail-form">
                <label>"Alt text "
                    <input type="text" prop:value=move || alt.get() on:input=move |ev| alt.set(event_target_value(&ev))/>
                </label>
                <label>"Caption "
                    <textarea rows=2 prop:value=move || caption.get() on:input=move |ev| caption.set(event_target_value(&ev))/>
                </label>
                <div class="detail-actions">
                    <button class="btn btn-primary" on:click=on_save>"Save"</button>
                    <span class="save-status">{move || save_status.get()}</span>
                </div>
            </div>
        </aside>
    }
}

fn human_bytes(n: u64) -> String {
    const KB: u64 = 1024; const MB: u64 = KB * 1024;
    if n >= MB { format!("{:.1} MB", n as f64 / MB as f64) }
    else if n >= KB { format!("{:.0} KB", n as f64 / KB as f64) }
    else { format!("{n} B") }
}
```
