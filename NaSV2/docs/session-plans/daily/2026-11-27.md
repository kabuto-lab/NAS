P2 · Save action

  `editor/save.rs` (NEW):

```rust
use leptos::prelude::*;
use leptos::task::spawn_local;
use serde::Serialize;

use crate::admin::context::AdminContext;
use crate::admin::editor::canvas::{EditorState, SaveStatus};

#[derive(Serialize)]
struct SavePayload<'a> {
    title: String,
    blocks: &'a [nas2_domain::Block],
}

pub fn save_now(state: EditorState, ctx: AdminContext, autosave: bool) {
    state.save_status.set(SaveStatus::Saving);
    spawn_local(async move {
        let url = format!(
            "{}/pages/{}{}",
            ctx.api_base,
            state.slug,
            if autosave { "?autosave=true" } else { "" }
        );
        let blocks = state.blocks.get_untracked();
        let payload = SavePayload {
            title: state.title.get_untracked(),
            blocks: &blocks,
        };
        let body = serde_json::to_string(&payload).unwrap_or_default();
        let req = web_sys::RequestInit::new();
        req.set_method("PATCH");
        req.set_body(&body.into());
        let headers = web_sys::Headers::new().unwrap();
        let _ = headers.set("Content-Type", "application/json");
        let _ = headers.set("X-CSRF-Token", &ctx.csrf_token);
        req.set_headers(&headers);
        let request = web_sys::Request::new_with_str_and_init(&url, &req).unwrap();
        let resp_promise = web_sys::window().unwrap().fetch_with_request(&request);
        let resp_js = wasm_bindgen_futures::JsFuture::from(resp_promise).await;
        match resp_js {
            Ok(resp_value) => {
                let resp: web_sys::Response = resp_value.dyn_into().unwrap();
                if resp.ok() {
                    if !autosave {
                        state.last_saved_at.set(Some(chrono::Utc::now()));
                    }
                    state.dirty.set(false);
                    state.save_status.set(SaveStatus::Saved);
                } else {
                    state.save_status.set(SaveStatus::Failed(
                        format!("server returned {}", resp.status())));
                }
            }
            Err(_) => state.save_status.set(SaveStatus::Failed("network error".into())),
        }
    });
}
```

  Note: requires `web-sys` + `wasm-bindgen-futures` features.
  Verify they're enabled in leptos workspace dep — they are
  (Leptos pulls them transitively for hydration).

```rust
fn SaveStatusBadge() -> impl IntoView {
    let state = expect_context::<EditorState>();
    view! {
        <span class="save-status">
            {move || match state.save_status.get() {
                SaveStatus::Idle => view! { <span class="muted">"Ready"</span> }.into_any(),
                SaveStatus::Saving => view! { <span>"Saving…"</span> }.into_any(),
                SaveStatus::Saved => view! {
                    <span class="ok">"Saved "
                        {state.last_saved_at.get().map(|t| t.format("%H:%M:%S").to_string())}
                    </span>
                }.into_any(),
                SaveStatus::Failed(msg) => view! {
                    <span class="error">"Save failed: " {msg}</span>
                }.into_any(),
            }}
        </span>
    }
}
```

```rust
use leptos::set_interval;
use std::time::Duration;

#[component]
pub fn PostEditor(post: Post) -> impl IntoView {
    let state = EditorState::from_post(post);
    let ctx = expect_context::<AdminContext>();
    provide_context(state.clone());

    // Autosave: every 30s, if dirty, fire autosave PATCH.
    let s = state.clone();
    let c = ctx.clone();
    let _handle = set_interval(
        move || {
            if s.dirty.get_untracked() && s.save_status.get_untracked() != SaveStatus::Saving {
                save::save_now(s.clone(), c.clone(), true);
            }
        },
        Duration::from_secs(30),
    );

    // ... rest of view
}
```

P6 · beforeunload guard

  Inline tiny JS shim:

```rust
view! {
    <PostEditor ... />
    <script>
        r#"
        window.addEventListener('beforeunload', (e) => {
          if (window.AX_EDITOR_DIRTY) {
            e.preventDefault();
            e.returnValue = '';
          }
        });
        // Hook for Leptos to update the flag.
        window.axSetDirty = (v) => { window.AX_EDITOR_DIRTY = v; };
```

PERFORMANCE / SECURITY / PITFALLS
═════════════════════════════════════════════════════════════════════

  - Autosave + manual save race: guard via `save_status != Saving`
    check before issuing. Worst case: two PATCHes land back-to-back
    on server; M3 W4 D5 ensures autosave revisions aren't created
    for empty diffs (deferred polish).
  - PATCH body can be large (blocks Vec) — JSON serialisation cost
    is per-save (~1KB for a typical post), negligible.
  - CSRF token from AdminContext stays fresh due to W1 D4 sliding
    refresh (token rotates on each cookie refresh).
  - beforeunload spec doesn't allow custom messages anymore — just
    triggers the browser default. Acceptable.
  - Cross-tab edit collision: if operator opens same post in two
    tabs and edits both, last save wins. Real conflict detection
    (ETag / optimistic concurrency) is a M9+ feature.
