P2 · `editor/canvas.rs` — `<PostEditor/>` and `<BlockCanvas/>`

```rust
use leptos::prelude::*;
use nas2_common::PostId;
use nas2_domain::{Block, Post};

use crate::admin::context::AdminContext;
use crate::admin::editor::node::BlockNode;

#[derive(Clone, Debug)]
pub struct EditorState {
    pub post_id: PostId,
    pub title: RwSignal<String>,
    pub blocks: RwSignal<Vec<Block>>,
    /// Editing mode index — Some(i) = block i is being edited;
    /// None = view-only state for all blocks.
    pub editing_index: RwSignal<Option<usize>>,
}

impl EditorState {
    pub fn from_post(p: Post) -> Self {
        Self {
            post_id: p.id,
            title: RwSignal::new(p.title),
            blocks: RwSignal::new(p.blocks),
            editing_index: RwSignal::new(None),
        }
    }
}

#[component]
pub fn PostEditor(post: Post) -> impl IntoView {
    let state = EditorState::from_post(post);
    provide_context(state.clone());
    let ctx = expect_context::<AdminContext>();

    view! {
        <div class="editor-shell">
            <header class="editor-toolbar">
                <input class="editor-title"
                       type="text"
                       prop:value=move || state.title.get()
                       on:input=move |ev| state.title.set(event_target_value(&ev))
                       placeholder="Post title"/>
                <div class="editor-actions">
                    <button class="btn">"Insert block"</button>
                    <button class="btn btn-primary">"Save"</button>
                </div>
            </header>
            <BlockCanvas/>
        </div>
    }
}

#[component]
pub fn BlockCanvas() -> impl IntoView {
    let state = expect_context::<EditorState>();
    view! {
        <ol class="block-canvas">
            <For
                each=move || state.blocks.get().into_iter().enumerate().collect::<Vec<_>>()
                key=|(i, b)| (*i, format!("{}", block_type_id(b)))
                children=move |(i, block)| view! {
                    <li class="block-row">
                        <BlockNode index=i block/>
                    </li>
                }
            />
        </ol>
    }
}

fn block_type_id(b: &Block) -> &'static str {
    use Block::*;
    match b {
        Heading { .. } => "heading",
        Paragraph { .. } => "paragraph",
        Image { .. } => "image",
        CodeBlock { .. } => "code_block",
        _ => "other",
    }
}
```

P3 · `editor/node.rs` — `<BlockNode/>` variant dispatcher

```rust
use leptos::prelude::*;
use nas2_domain::Block;

use crate::admin::editor::canvas::EditorState;
use crate::admin::editor::variants::{heading, paragraph, image, code_block, placeholder};

#[component]
pub fn BlockNode(index: usize, block: Block) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let is_editing = Memo::new(move |_| state.editing_index.get() == Some(index));

    view! {
        <article class="block-node"
                 class:editing=is_editing
                 data-index=index.to_string()>
            <aside class="block-toolbar">
                {/* D2 will add: insert-above, edit, move-up, move-down, delete */}
            </aside>
            <div class="block-body">
                {match block {
                    Block::Heading { level, text } =>
                        view! { <heading::HeadingView level text/> }.into_any(),
                    Block::Paragraph { html } =>
                        view! { <paragraph::ParagraphView html/> }.into_any(),
                    Block::Image { src, alt, caption } =>
                        view! { <image::ImageView src alt caption/> }.into_any(),
                    Block::CodeBlock { lang, code } =>
                        view! { <code_block::CodeBlockView lang code/> }.into_any(),
                    other =>
                        view! { <placeholder::PlaceholderView block=other/> }.into_any(),
                }}
            </div>
        </article>
    }
}
```

```rust
use leptos::prelude::*;
use nas2_domain::HeadingLevel;

#[component]
pub fn HeadingView(level: HeadingLevel, text: String) -> impl IntoView {
    let tag = format!("h{}", level.as_u8());
    let label = format!("H{}", level.as_u8());
    view! {
        <div class="block-heading">
            <span class="variant-badge">{label}</span>
            // Render the actual heading element with auto-escape.
            // Use dynamic-tag pattern: Leptos 0.7 has `Element::dyn`.
            // Simpler: match level and render the right tag inline.
            {match level {
                HeadingLevel::H1 => view! { <h1>{text.clone()}</h1> }.into_any(),
                HeadingLevel::H2 => view! { <h2>{text.clone()}</h2> }.into_any(),
                HeadingLevel::H3 => view! { <h3>{text.clone()}</h3> }.into_any(),
                HeadingLevel::H4 => view! { <h4>{text.clone()}</h4> }.into_any(),
                HeadingLevel::H5 => view! { <h5>{text.clone()}</h5> }.into_any(),
                HeadingLevel::H6 => view! { <h6>{text.clone()}</h6> }.into_any(),

}
```

```rust
#[component]
pub fn ParagraphView(html: String) -> impl IntoView {
    // SAFETY: Block::Paragraph.html is GUARANTEED sanitize-on-write
```

P5 · Handler `admin/posts.rs::post_edit`

```rust
// caps.require(cms.page.draft)
pub async fn post_edit(
    _: RequireCaps<CmsPageDraft>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Extension(csrf): Extension<CsrfToken>,
    Extension(user_id): Extension<UserId>,
    Path(slug_raw): Path<String>,
) -> Result<impl IntoResponse, AppError> {
    let slug = nas2_domain::PostSlug::try_new(&slug_raw)?;
    let post = state.post_repo.find_by_slug(tctx.tenant_id, tctx.site_id, &slug).await?
        .ok_or_else(|| AppError::NotFound(format!("post {slug_raw}")))?;
    let ctx = build_admin_ctx(/* ... */);
    let crumbs = BreadcrumbChain::root()
        .push(Crumb::linked("Posts", "/admin/posts"))
        .push(Crumb::terminal(post.title.clone()));

    let html = leptos::ssr::render_to_string(move || view! {
        <AdminLayout ctx title="Edit Post" crumbs>
            <PostEditor post/>
        </AdminLayout>
    });
    Ok(Html(html.into_owned()))
}
```

PERFORMANCE / SECURITY / PITFALLS
═════════════════════════════════════════════════════════════════════

  - **CRITICAL: inner_html on Paragraph.** The domain invariant is
    "sanitize on write" (§3.6, M2 W8 D4). If a future code path
    bypasses ammonia and writes raw user HTML into Block::Paragraph,
    today's renderer would surface XSS. Mitigation: add a defensive
    `ammonia::clean()` call at PostRepository::insert/update in M4
    W3's validation pass (verify the M4 work covered this — if not,
    log as carry-over).
  - <For> with stable keys minimizes re-renders. Today's key is
    `(index, variant_id)` — both stable per render but the SAME
    pair for a given block; that's fine for view-only.
  - Image `loading="lazy"` for cheap lazy-load.
  - Per-block hover toolbar visibility via CSS — no JS, no perf cost.
  - 15 placeholder variants: today's "[Variant: X]" rendering is
    semantically accessible; will be replaced incrementally in M8+
    with real per-variant editors.
