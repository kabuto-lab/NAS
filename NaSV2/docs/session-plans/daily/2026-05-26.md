  ```rust
  //! `xtask check-planning-refs` · ENTITY §13 planning trail gate.
  //!
  //! Walks `git log <range> --format=%H%n%B%n--END--` and verifies
  //! each non-trivial commit message contains at least one
  //! `(RFC|ADR|PLAN|VAL)-\d{1,4}` reference. Trivial commits
  //! (typo / dep bump / fmt / comment) are exempt per ENTITY §13.
  //!
  //! ## Severity
  //!
  //! Exit 1 on any violation. Designed to gate PRs in
  //! `.github/workflows/nasv2-ci.yml` (job `xtask-gates`).
  //!
  //! ## TLA layers
  //!
  //! - L1 Correctness: regex tested against canonical examples
  //! - L2 Performance: single `git log` invocation; in-process scan
  //! - L3 Scalability: bounded by commit count in range; CI typically
  //!   passes `--since main~10` or compares against the merge-base
  //! - L4 Operability: violations printed with SHA + first line + reason

  use std::process::Command;

  use eyre::{eyre, Result, WrapErr};
  use regex::Regex;

  pub fn run(commit: &str) -> Result<()> {
      let messages = git_log_messages(commit)?;
      let ref_re = Regex::new(r"\b(?:RFC|ADR|PLAN|VAL)-\w{1,16}\b").unwrap();
      let trivial_re = trivial_commit_regex();
      let mut violations: Vec<Violation> = Vec::new();

      for entry in messages {
          let first_line = entry.body.lines().next().unwrap_or("").to_owned();
          if trivial_re.is_match(&first_line) {
              println!("OK (trivial): {}  {}", &entry.sha[..7], first_line);
              continue;
          }
          if ref_re.is_match(&entry.body) {
              println!("OK         : {}  {}", &entry.sha[..7], first_line);
              continue;
          }
          violations.push(Violation {
              sha: entry.sha.clone(),
              first_line,
              reason: "no RFC/ADR/PLAN/VAL ref and not trivial",
          });
      }

      if violations.is_empty() {
          println!("xtask check-planning-refs: ok ({} commit(s) checked)", messages.len());
          return Ok(());
      }
      for v in &violations {
          eprintln!("ERROR: {sha}  {first}  ({reason})  [ENTITY §13]",
              sha = &v.sha[..7], first = v.first_line, reason = v.reason);
      }
      Err(eyre!("xtask check-planning-refs: {} violation(s)", violations.len()))
  }

  // ── git log harvest ───────────────────────────────────────────────────

  struct CommitEntry {
      sha: String,
      body: String,
  }

  fn git_log_messages(range_spec: &str) -> Result<Vec<CommitEntry>> {
      // %x00 between SHA and body, %x01 between commits — chosen to be
      // bytes that never appear in legitimate commit messages.
      let out = Command::new("git")
          .args(["log", range_spec, "--format=%H%x00%B%x01"])
          .output()
          .wrap_err("invoking git log — is git in PATH?")?;
      if !out.status.success() {
          return Err(eyre!(
              "git log {range_spec} failed: {}",
              String::from_utf8_lossy(&out.stderr)
          ));
      }
      let raw = String::from_utf8(out.stdout).wrap_err("git log output not UTF-8")?;
      let mut entries = Vec::new();
      for chunk in raw.split('\x01').filter(|s| !s.trim().is_empty()) {
          let mut parts = chunk.splitn(2, '\x00');
          let sha = parts.next().unwrap_or("").trim().to_owned();
          let body = parts.next().unwrap_or("").to_owned();
          if !sha.is_empty() {
              entries.push(CommitEntry { sha, body });
          }
      }
      Ok(entries)
  }

  // ── trivial-commit allow-list ─────────────────────────────────────────

  fn trivial_commit_regex() -> Regex {
      Regex::new(concat!(
          r"^(?:",
          // typo fixes
          r"typo[:(]",
          r"|fix:? typo",
          // chore: fmt / formatting / lint
          r"|chore[(:][^)]*\)?\s*fmt",
          r"|style:? format",
          // dep bumps
          r"|chore[(:][^)]*\)?\s*deps?",
          r"|build:? bump",
          // comment-only
          r"|docs?[:(]? comment",
          r"|comment:",
          r")",
      )).unwrap()
  }

  // ── violation ─────────────────────────────────────────────────────────

  struct Violation {
      sha: String,
      first_line: String,
      reason: &'static str,
  }
  ```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`git` not in PATH.** Wrap with explicit message — done in
     `git_log_messages`. Windows installs sometimes ship git outside
     PATH; if so, HARD STOP and tell operator to fix env.
  2. **Body delimiter collision.** `\x00` and `\x01` chosen because
     they're invalid in commit messages (git refuses them). If a
     future hook starts injecting them, switch to a UUID-derived
     delimiter.
  3. **Range spec acceptance.** `git log` accepts both `HEAD`,
     `HEAD~5..HEAD`, `main..HEAD`, and bare SHAs. The clap arg has
     `default_value = "HEAD"` already (see xtask/src/main.rs).
  4. **Merge commits.** `--format=%B` reports merge commit messages
     same as regular ones. Acceptable — merge messages should still
     carry refs from the merged branch's tip.
  5. **Trivial regex over-matches.** `chore: fmt` matches
     `chore: fmt + add gizmo` — that's the design (operator chose
     "this is fmt-driven so I don't have to write a ref"). If
     abuse appears in retro, tighten regex.
