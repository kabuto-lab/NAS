//! Password verification — argon2id.
//!
//! Hashing happens in the admin-create-user flow (M2+) AND in
//! `nas2-cli user create` (deferred). Verification happens at login.

use argon2::{Argon2, password_hash::{PasswordHash, PasswordVerifier}};
use nas2_common::AppError;

/// Returns Ok(true) if the password matches the encoded hash.
/// `hash` is the encoded argon2 string `$argon2id$v=19$m=...`.
#[must_use = "result indicates whether auth succeeded"]
pub fn verify_password(plain: &str, hash: &str) -> Result<bool, AppError> {
    let parsed = PasswordHash::new(hash)
        .map_err(|e| {
            tracing::error!(error=%e, "argon2 hash parse");
            AppError::Internal
        })?;
    match Argon2::default().verify_password(plain.as_bytes(), &parsed) {
        Ok(()) => Ok(true),
        Err(argon2::password_hash::Error::Password) => Ok(false),
        Err(e) => {
            tracing::error!(error=%e, "argon2 verify");
            Err(AppError::Internal)
        }
    }
}

//! `PgUserRepository` — SQLx adapter for the `users` table.

use async_trait::async_trait;
use nas2_application::ports::UserRepository;
use nas2_common::{AppError, TenantId, UserId};
use nas2_domain::{Email, User};
use nas2_tenant::with_tenant;
use sqlx::{postgres::PgRow, FromRow, PgPool, Row};

#[derive(Clone)]
pub struct PgUserRepository {
    pool: PgPool,
}

impl PgUserRepository {
    #[must_use]
    pub const fn new(pool: PgPool) -> Self { Self { pool } }

    /// Internal method (not on the port) — returns the password hash
    /// alongside the user. Login path needs this; nothing else does.
    pub async fn find_by_email_with_hash(
        &self,
        tenant: TenantId,
        email: &Email,
    ) -> Result<Option<(User, String)>, AppError> {
        let email_str = email.as_str().to_owned();
        with_tenant(&self.pool, tenant, |tx| async move {
            let row: Option<(uuid::Uuid, uuid::Uuid, String, String,
                              Option<String>, chrono::DateTime<chrono::Utc>)> =
                sqlx::query_as(
                    "SELECT id, tenant_id, email, password_hash, display_name, created_at \
                     FROM users WHERE email = $1::citext"
                )
                .bind(&email_str)
                .fetch_optional(&mut **tx).await
                .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
            match row {
                None => Ok(None),
                Some((id, tid, em, hash, _name, created_at)) => {
                    let email = Email::try_new(&em)?;
                    let user = User {
                        id: UserId::from_uuid(id),
                        tenant_id: TenantId::from_uuid(tid),
                        email,
                        role_ids: Vec::new(), // loaded separately if needed
                        created_at,
                    };
                    Ok(Some((user, hash)))
                }
            }
        }).await
    }
}

#[async_trait]
impl UserRepository for PgUserRepository {
    async fn find_by_id(
        &self,
        tenant: TenantId,
        id: UserId,
    ) -> Result<Option<User>, AppError> {
        let uid = id.into_uuid();
        with_tenant(&self.pool, tenant, |tx| async move {
            let row: Option<(uuid::Uuid, uuid::Uuid, String,
                              Option<String>, chrono::DateTime<chrono::Utc>)> =
                sqlx::query_as(
                    "SELECT id, tenant_id, email, display_name, created_at \
                     FROM users WHERE id = $1"
                ).bind(uid).fetch_optional(&mut **tx).await
                 .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
            match row {
                None => Ok(None),
                Some((id, tid, em, _name, created_at)) => {
                    let email = Email::try_new(&em)?;
                    Ok(Some(User {
                        id: UserId::from_uuid(id),
                        tenant_id: TenantId::from_uuid(tid),
                        email,
                        role_ids: Vec::new(),
                        created_at,
                    }))
                }
            }
        }).await
    }

    async fn find_by_email(
        &self,
        tenant: TenantId,
        email: &Email,
    ) -> Result<Option<User>, AppError> {
        Ok(self.find_by_email_with_hash(tenant, email).await?.map(|(u, _)| u))
    }

    async fn insert(&self, _user: &User) -> Result<(), AppError> {
        // The plain `insert` on the trait doesn't carry password —
        // login-relevant inserts go through a separate command in M3+
        // (admin user-create flow). For now: stub returning Internal
        // so nothing accidentally calls it.
        Err(AppError::Internal)
    }

    async fn update(&self, _user: &User) -> Result<(), AppError> {
        Err(AppError::Internal /* M3+ command */)
    }
}

#![allow(dead_code)]

use argon2::{Argon2, password_hash::{rand_core::OsRng, PasswordHasher, SaltString}};

pub fn hash_password(plain: &str) -> String {
    let salt = SaltString::generate(&mut OsRng);
    Argon2::default()
        .hash_password(plain.as_bytes(), &salt)
        .expect("argon2 hash")
        .to_string()
}

═════════════════════════════════════════════════════════════════════
COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`argon2::default()` params** (m=19456, t=2, p=1) take ~100 ms
     per verify. Acceptable for login but DON'T put on hot read paths.
     M3+ can tune the params via env var.
  2. **CITEXT comparison in SQL** — sqlx binds as TEXT; cast in SQL
     with `$1::citext` for case-insensitive match.
  3. **Password hash leaves the application layer ONLY via
     `find_by_email_with_hash`.** The trait `UserRepository` has no
     hash. Login pipeline calls the concrete repo's helper directly.
  4. **argon2id encoded format** is self-describing — it carries
     params, salt, hash in one string. Storing as TEXT is fine.
  5. **`UserRepository::insert/update` stub** prevents accidental
     production callers from creating users without password —
     M3 has the proper command shape.
