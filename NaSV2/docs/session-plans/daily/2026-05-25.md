  ```rust
  //! `xtask magic-check` · forbid runtime footguns by grep · ENTITY §18.2
  //!
  //! Rules enforced (each via a `Rule` struct so the set is extensible):
  //!   R1  raw `tokio::spawn(` outside `crates/runtime/src/supervisor.rs`
  //!       and dev-test files that opt in via file-scope
  //!       `#![allow(clippy::disallowed_methods)]`
  //!   R2  `lazy_static!` outside `crates/common/src/registry/` (not yet
  //!       extant — rule armed for the future)
  //!   R3  `macro_rules!` definitions whose body exceeds 50 lines —
  //!       per §9.4, magic-heavy macros are forbidden
  //!
  //! Note: this gate complements clippy. clippy `disallowed-methods`
  //! covers compile-time use; magic-check catches violations even when
  //! someone bypassed clippy locally, and runs across .rs files
  //! whether or not they are in a `cargo check` target set.

  use std::path::{Path, PathBuf};

  use eyre::{eyre, Result, WrapErr};
  use regex::Regex;
  use walkdir::WalkDir;

  /// Skipped path fragments. `walkdir` is fed `crates/` and `apps/`;
  /// these prefixes prune large generated directories.
  const SKIP_FRAGMENTS: &[&str] = &[
      // build artifacts
      "target",
      // dep snapshots
      ".sqlx",
      // VCS
      ".git",
      // editor noise
      ".vscode",
      ".idea",
  ];

  /// Paths (relative to workspace root) where raw `tokio::spawn` is OK.
  const SPAWN_WHITELIST: &[&str] = &[
      "crates/runtime/src/supervisor.rs",
  ];

  pub fn run() -> Result<()> {
      let root = workspace_root()?;
      let scan_dirs = ["crates", "apps", "xtask"]; // not migrations/docs
      let mut violations: Vec<Violation> = Vec::new();

      let rules = [
          Rule::new(R1_NAME, build_spawn_regex())
              .with_path_whitelist(SPAWN_WHITELIST)
              .with_file_scope_allow("clippy::disallowed_methods"),
          Rule::new(R2_NAME, Regex::new(r"\blazy_static!\s*\{").unwrap()),
          // R3 handled separately — line-counting, not pure regex
      ];

      for dir in scan_dirs {
          let abs = root.join(dir);
          if !abs.is_dir() { continue; }
          for entry in WalkDir::new(&abs).into_iter().filter_map(|e| e.ok()) {
              if !is_rust_source(entry.path()) { continue; }
              if path_should_skip(entry.path()) { continue; }
              let rel = entry.path().strip_prefix(&root)
                  .unwrap_or(entry.path()).to_path_buf();
              let body = std::fs::read_to_string(entry.path())
                  .wrap_err_with(|| format!("read {}", rel.display()))?;
              for rule in &rules {
                  rule.scan(&rel, &body, &mut violations);
              }
              scan_macro_size(&rel, &body, &mut violations);
          }
      }

      if violations.is_empty() {
          println!("xtask magic-check: ok ({} scanned)", scan_dirs.len());
          return Ok(());
      }
      for v in &violations {
          eprintln!("ERROR: {v}");
      }
      Err(eyre!("xtask magic-check: {} violation(s) — see stderr", violations.len()))
  }

  // ── R1/R2 generic engine ──────────────────────────────────────────────

  const R1_NAME: &str = "tokio_spawn";
  const R2_NAME: &str = "lazy_static";
  const R3_NAME: &str = "macro_rules_50_lines";

  struct Rule {
      name: &'static str,
      pattern: Regex,
      path_whitelist: &'static [&'static str],
      file_scope_allow: Option<&'static str>,
  }
  impl Rule {
      fn new(name: &'static str, pattern: Regex) -> Self {
          Self { name, pattern, path_whitelist: &[], file_scope_allow: None }
      }
      fn with_path_whitelist(mut self, w: &'static [&'static str]) -> Self {
          self.path_whitelist = w;
          self
      }
      fn with_file_scope_allow(mut self, lint: &'static str) -> Self {
          self.file_scope_allow = Some(lint);
          self
      }
      fn scan(&self, rel: &Path, body: &str, out: &mut Vec<Violation>) {
          let rel_str = rel.to_string_lossy().replace('\\', "/");
          if self.path_whitelist.iter().any(|w| rel_str == *w) { return; }
          // file-scope allow opt-out (test files)
          if let Some(lint) = self.file_scope_allow {
              let needle = format!("#![allow({lint}");
              if body.contains(&needle) { return; }
          }
          for (lineno, line) in body.lines().enumerate() {
              // Skip lines inside line-comments and inside doc-comments.
              // Block comments (/* */) are not handled — keep the rule
              // imperfect rather than complex; macro_rules! big-body
              // catches one form of magic.
              if is_doc_or_line_comment(line) { continue; }
              if self.pattern.is_match(line) {
                  out.push(Violation {
                      rule: self.name,
                      file: rel.to_path_buf(),
                      line: lineno + 1,
                      excerpt: line.trim().to_owned(),
                  });
              }
          }
      }
  }

  fn build_spawn_regex() -> Regex {
      // Word-boundary anchored; `tokio::spawn(` MUST appear with the
      // open paren. Bare `spawn(` (imported) is also caught via the
      // optional `tokio::` prefix.
      Regex::new(r"\btokio::spawn\s*\(").unwrap()
  }

  fn is_doc_or_line_comment(line: &str) -> bool {
      let trimmed = line.trim_start();
      trimmed.starts_with("//") || trimmed.starts_with("///")
          || trimmed.starts_with("//!")
  }

  fn is_rust_source(p: &Path) -> bool {
      p.is_file() && p.extension().and_then(|e| e.to_str()) == Some("rs")
  }

  fn path_should_skip(p: &Path) -> bool {
      let s = p.to_string_lossy();
      SKIP_FRAGMENTS.iter().any(|f| s.contains(f))
  }

  fn workspace_root() -> Result<PathBuf> {
      let manifest = std::env::var("CARGO_MANIFEST_DIR")
          .wrap_err("CARGO_MANIFEST_DIR not set — run via `cargo xtask`")?;
      let root = Path::new(&manifest).parent()
          .ok_or_else(|| eyre!("xtask manifest has no parent"))?
          .to_path_buf();
      Ok(root)
  }

  // ── R3: macro_rules! body length ──────────────────────────────────────

  const MACRO_BODY_MAX_LINES: usize = 50;

  fn scan_macro_size(rel: &Path, body: &str, out: &mut Vec<Violation>) {
      // Heuristic — count lines between `macro_rules! NAME {` and the
      // matching closing `}` at column 0. Imperfect for nested braces
      // but acceptable as a regression alarm; the day a 50-line macro
      // wants in, an explicit waiver via comment will be added.
      let mut depth: i32 = 0;
      let mut start_line: Option<usize> = None;
      let macro_open = Regex::new(r"^\s*macro_rules!\s+\w+\s*\{").unwrap();
      for (lineno, line) in body.lines().enumerate() {
          if start_line.is_none() && macro_open.is_match(line) {
              start_line = Some(lineno);
              depth = 1;
              continue;
          }
          if start_line.is_some() {
              depth += line.matches('{').count() as i32;
              depth -= line.matches('}').count() as i32;
              if depth <= 0 {
                  let body_lines = lineno - start_line.unwrap();
                  if body_lines > MACRO_BODY_MAX_LINES {
                      out.push(Violation {
                          rule: R3_NAME,
                          file: rel.to_path_buf(),
                          line: start_line.unwrap() + 1,
                          excerpt: format!(
                              "macro_rules! body = {body_lines} lines (limit {MACRO_BODY_MAX_LINES})"
                          ),
                      });
                  }
                  start_line = None;
              }
          }
      }
  }

  // ── Violation type ────────────────────────────────────────────────────

  struct Violation {
      rule: &'static str,
      file: PathBuf,
      line: usize,
      excerpt: String,
  }
  impl std::fmt::Display for Violation {
      fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
          write!(f, "{rule} at {file}:{line}: {excerpt}  [ENTITY §18.2]",
              rule = self.rule, file = self.file.display(),
              line = self.line, excerpt = self.excerpt)
      }
  }
  ```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`walkdir` traverses target/.** SKIP_FRAGMENTS prunes it but the
     check is `s.contains(f)` — make sure `target` isn't a substring of
     a legitimate path component. None in our tree, but flag if it
     appears.
  2. **CRLF vs LF.** Windows hosts (this one) inflate `line.lines()`
     newline behaviour — verify `line.lines()` strips both — it does
     per stdlib docs.
  3. **`tokio::spawn` in doc-strings.** The `is_doc_or_line_comment`
     check is line-prefix only; a `tokio::spawn` inside a `/* ... */`
     block comment WILL trigger. Accept as known limitation; flag in
     code comment.
  4. **Macro body counter** treats `}` inside string literals as
     real braces. Won't happen in our codebase today; document as
     known limitation.
  5. **`std::env::var("CARGO_MANIFEST_DIR")`** is set by Cargo when
     `cargo xtask` is invoked; bare `cargo run -p xtask --
     magic-check` also sets it. Manual `./target/debug/xtask` does
     NOT — `wrap_err` produces a clear message.
