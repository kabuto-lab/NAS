```rust
use leptos::prelude::*;
use nas2_common::MediaId;

#[component]
pub fn MediaPickerModal(
    on_pick: Callback<MediaTile>,
    on_close: Callback<()>,
    #[prop(default = Some("image/"))] mime_prefix: Option<&'static str>,
) -> impl IntoView {
    let items = RwSignal::new(Vec::<MediaTile>::new());
    let loaded = RwSignal::new(false);

    spawn_local({
        let items = items.clone();
        let loaded = loaded.clone();
        async move {
            // Fetch first page filtered by mime_prefix

            if let Ok(page) = resp { items.set(page.items); }
            loaded.set(true);
        }
    });

    view! {
        <div class="modal-overlay" role="dialog" aria-label="media picker">
            <div class="modal-backdrop" on:click=move |_| on_close.call(())></div>
            <div class="modal-content media-picker">
                <header>
                    <h3>"Choose image"</h3>
                    <button class="modal-close" on:click=move |_| on_close.call(())>"×"</button>
                </header>
                <Show when=move || loaded.get() fallback=|| view! { <p>"Loading…"</p> }>
                    <ul class="media-grid">
                        <For each=move || items.get()
                             key=|m| m.id
                             children=move |m| {
                                 let pick = on_pick.clone();
                                 view! {
                                     <li class="media-tile picker"
                                         on:click=move |_| pick.call(m.clone())>
                                         <div class="media-thumb">
                                             <img src=m.thumb_url alt=m.alt/>
                                         </div>
                                     </li>
                                 }
                             }/>
                    </ul>
                </Show>

        </div>
    }
}
```

P3 · Drag-drop upload onto canvas

  `crates/presentation/src/admin/editor/dnd_upload.rs` (NEW):

```rust
use leptos::prelude::*;
use leptos::task::spawn_local;
use nas2_domain::Block;
use web_sys::{DragEvent, FormData};

use crate::admin::context::AdminContext;
use crate::admin::editor::canvas::EditorState;

#[component]
pub fn DropZone(children: Children) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let ctx = expect_context::<AdminContext>();
    let drag_active = RwSignal::new(false);
    let upload_progress = RwSignal::new(None::<(String, u32)>);

    let on_dragover = move |ev: DragEvent| {
        ev.prevent_default();
        drag_active.set(true);
    };
    let on_dragleave = move |_: DragEvent| drag_active.set(false);
    let on_drop = move |ev: DragEvent| {
        ev.prevent_default();
        drag_active.set(false);
        let Some(files) = ev.data_transfer().and_then(|dt| dt.files()) else { return; };
        for i in 0..files.length() {
            if let Some(file) = files.item(i) {
                let s = state.clone();
                let c = ctx.clone();
                let pg = upload_progress.clone();
                spawn_local(async move {
                    upload_file(s, c, file, pg).await;
                });
            }
        }
    };

    view! {
        <div class="dropzone"
             class:active=drag_active
             on:dragover=on_dragover
             on:dragleave=on_dragleave
             on:drop=on_drop>
            <Show when=move || drag_active.get() fallback=|| ()>
                <div class="dropzone-overlay">"Drop to upload"</div>
            </Show>
            <Show when=move || upload_progress.get().is_some() fallback=|| ()>
                <div class="upload-progress">
                    {move || upload_progress.get().map(|(name, pct)|
                        format!("Uploading {name}… {pct}%"))}
                </div>
            </Show>
            {children()}
        </div>
    }
}

async fn upload_file(
    state: EditorState,
    ctx: AdminContext,
    file: web_sys::File,
    progress: RwSignal<Option<(String, u32)>>,
) {
    let name = file.name();
    progress.set(Some((name.clone(), 0)));

    let form = FormData::new().unwrap();
    let _ = form.append_with_blob("file", &file);

    let req = web_sys::RequestInit::new();
    req.set_method("POST");
    req.set_body(&form.into());
    let headers = web_sys::Headers::new().unwrap();
    let _ = headers.set("X-CSRF-Token", &ctx.csrf_token);
    req.set_headers(&headers);
    let url = format!("{}/media", ctx.api_base);
    let request = web_sys::Request::new_with_str_and_init(&url, &req).unwrap();
    let resp_promise = web_sys::window().unwrap().fetch_with_request(&request);
    let resp_js = wasm_bindgen_futures::JsFuture::from(resp_promise).await;
    progress.set(Some((name.clone(), 100)));

    if let Ok(resp_value) = resp_js {
        let resp: web_sys::Response = resp_value.dyn_into().unwrap();
        if resp.ok() {
            // Parse response { id, url, ... } → insert Image block.
            // Simplified: just insert a placeholder image with the
            // returned URL. Full impl parses JSON.
            let json_promise = resp.json().unwrap();
            if let Ok(json) = wasm_bindgen_futures::JsFuture::from(json_promise).await {
                // Extract src + alt from response
                // (would parse via serde-wasm-bindgen — non-trivial inline)
                state.blocks.update(|bs| {
                    bs.push(Block::Image {
                        src: "/uploaded.jpg".into(),  // placeholder — parse real value
                        alt: name.clone(),
                        caption: None,
                    });
                });
                state.mark_dirty();
            }
        }
    }
    // Clear progress after 1.5s
    leptos::set_timeout(move || progress.set(None), std::time::Duration::from_millis(1500));
}
```
