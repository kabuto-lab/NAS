//! `PgPostRepository` — SQLx adapter for the `posts` table.

use async_trait::async_trait;
use nas2_application::ports::PostRepository;
use nas2_common::{AppError, Page, PostId, SiteId, TenantId};
use nas2_domain::{Block, CustomPostType, Post, PostSlug, PostStatus};
use nas2_tenant::with_tenant;
use sqlx::{postgres::PgRow, FromRow, PgPool, Row};

#[derive(Clone)]
pub struct PgPostRepository {
    pool: PgPool,
}

impl PgPostRepository {
    #[must_use]
    pub const fn new(pool: PgPool) -> Self { Self { pool } }
}

// Private row → domain mapping.
struct PostRow {
    id: uuid::Uuid,
    site_id: uuid::Uuid,
    tenant_id: uuid::Uuid,
    slug: String,
    title: String,
    status: String,
    published_at: Option<chrono::DateTime<chrono::Utc>>,
    blocks_json: serde_json::Value,
    custom_type: Option<String>,
}

impl PostRow {
    fn into_domain(self) -> Result<Post, AppError> {
        let status = match self.status.as_str() {
            "draft"     => PostStatus::Draft,
            "scheduled" => PostStatus::Scheduled,
            "published" => PostStatus::Published,
            "archived"  => PostStatus::Archived,
            other => return Err(AppError::Internal /* log */),
        };
        let blocks: Vec<Block> = serde_json::from_value(self.blocks_json)
            .map_err(|e| {
                tracing::error!(error=%e, "deserializing posts.blocks");
                AppError::Internal
            })?;
        Ok(Post {
            id: PostId::from_uuid(self.id),
            site_id: SiteId::from_uuid(self.site_id),
            tenant_id: TenantId::from_uuid(self.tenant_id),
            slug: PostSlug::try_new(&self.slug)?,
            title: self.title,
            status,
            published_at: self.published_at,
            blocks,
            custom_type: self.custom_type.map(CustomPostType),
        })
    }
}

impl<'r> FromRow<'r, PgRow> for PostRow {
    fn from_row(row: &'r PgRow) -> Result<Self, sqlx::Error> {
        Ok(Self {
            id: row.try_get("id")?,
            site_id: row.try_get("site_id")?,
            tenant_id: row.try_get("tenant_id")?,
            slug: row.try_get("slug")?,
            title: row.try_get("title")?,
            status: row.try_get("status")?,
            published_at: row.try_get("published_at")?,
            blocks_json: row.try_get("blocks")?,
            custom_type: row.try_get("custom_type")?,
        })
    }
}

#[async_trait]
impl PostRepository for PgPostRepository {
    async fn find_by_slug(
        &self,
        tenant: TenantId,
        site: SiteId,
        slug: &PostSlug,
    ) -> Result<Option<Post>, AppError> {
        let slug = slug.as_str().to_owned();
        let site_uuid = site.into_uuid();
        with_tenant(&self.pool, tenant, |tx| async move {
            let row: Option<PostRow> = sqlx::query_as(
                "SELECT id, site_id, tenant_id, slug, title, status, \
                        published_at, blocks, custom_type \
                 FROM posts \
                 WHERE site_id = $1 AND slug = $2"
            )
                .bind(site_uuid)
                .bind(&slug)
                .fetch_optional(&mut **tx).await
                .map_err(|e| {
                    tracing::error!(error=%e, "find_by_slug query");
                    AppError::Internal
                })?;
            match row {
                Some(r) => Ok(Some(r.into_domain()?)),
                None => Ok(None),
            }
        }).await
    }

    async fn find_by_id(
        &self,
        tenant: TenantId,
        id: PostId,
    ) -> Result<Option<Post>, AppError> {
        let id_uuid = id.into_uuid();
        with_tenant(&self.pool, tenant, |tx| async move {
            let row: Option<PostRow> = sqlx::query_as(
                "SELECT id, site_id, tenant_id, slug, title, status, \
                        published_at, blocks, custom_type \
                 FROM posts \
                 WHERE id = $1"
            )
                .bind(id_uuid)
                .fetch_optional(&mut **tx).await
                .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
            match row {
                Some(r) => Ok(Some(r.into_domain()?)),
                None => Ok(None),
            }
        }).await
    }

    // list_published, insert, update implemented in W6 D3/D4.

    }

    async fn update(&self, _post: &Post) -> Result<(), AppError> {

    }
}

    testcontainers::ContainerAsync<Postgres>,
    PgPostRepository,
    sqlx::PgPool,

    let pg = Postgres::default()
        .with_user("postgres").with_password("postgres").with_db_name("ax")
        .start().await.unwrap();
    let port = pg.get_host_port_ipv4(5432).await.unwrap();
    let url = format!("postgres://postgres:postgres@127.0.0.1:{port}/ax");
    let pool = PgPoolOptions::new().connect(&url).await.unwrap();
    for n in 2..=7 {
        let p = glob::glob(&format!("../../migrations/{n:04}_*.sql"))
            .unwrap().filter_map(Result::ok).next().unwrap();
        let sql = std::fs::read_to_string(&p).unwrap();
        sqlx::query(&sql).execute(&pool).await.unwrap();
    }
    let repo = PgPostRepository::new(pool.clone());
    (pg, repo, pool)
}

async fn seed_post(pool: &sqlx::PgPool, tenant: TenantId, site_id: SiteId,
                    slug: &str, status: &str) -> PostId {
    let post_id = PostId::new_v4();
    with_tenant(pool, tenant, |tx| async move {
        sqlx::query("SET LOCAL ROLE ax_app").execute(&mut **tx).await.unwrap();
        sqlx::query(
            "INSERT INTO sites (id, tenant_id, slug, display_name) \
             VALUES ($1, $2, 's', 'S') ON CONFLICT DO NOTHING"
        ).bind(site_id.into_uuid()).bind(tenant.into_uuid())
         .execute(&mut **tx).await.unwrap();
        sqlx::query(
            "INSERT INTO posts (id, site_id, tenant_id, slug, title, status, \
                                 published_at, blocks) \
             VALUES ($1, $2, $3, $4, 'T', $5, $6, '[]'::jsonb)"
        ).bind(post_id.into_uuid()).bind(site_id.into_uuid())
         .bind(tenant.into_uuid()).bind(slug).bind(status)
         .bind(if status == "published" { Some(Utc::now()) } else { None })
         .execute(&mut **tx).await.unwrap();
        Ok::<_, nas2_common::AppError>(())
    }).await.unwrap();
    post_id
}

#[tokio::test]
#[ignore = "needs Docker"]
async fn find_by_slug_returns_post_for_correct_tenant() {
    let (_pg, repo, pool) = seed_pool_and_repo().await;
    let tenant = TenantId::new_v4();
    let site = SiteId::new_v4();
    seed_post(&pool, tenant, site, "hello", "published").await;

    let slug = PostSlug::try_new("hello").unwrap();
    let got = repo.find_by_slug(tenant, site, &slug).await.unwrap();

    assert_eq!(got.unwrap().slug.as_str(), "hello");

}

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`&mut **tx` double-deref.** `tx` is `&mut Transaction`,
     deref to `&mut PgConnection`. sqlx `execute(...)` accepts
     `&mut PgConnection`. Pattern: `execute(&mut **tx)`.
  2. **`status: String` from PG TEXT vs domain enum.** Mapping
     bounded — keep the match arms in `PostRow::into_domain` in sync
     with `PostStatus`'s variants. Add a CHECK constraint test if
     drift becomes a worry.
  3. **`blocks_json: serde_json::Value` deserialization.** Domain
     `Vec<Block>` derives serde; JSONB column passes through. If a
     row has a Block variant unknown to the current binary, deser
     fails — rolling deploys should land NEW Block variants in code
     BEFORE any writes use them.
  4. **`sqlx::query_as` requires `FromRow` impl** — we provide one
     explicitly. Could derive via `#[derive(FromRow)]` but the
     manual impl is clearer for the `blocks_json` JSONB column.
  5. **Logging via `tracing::error!` + returning `AppError::Internal`**
     — gives operators the detail (DB error message) without leaking
     to client. Pattern repeats in W6 D3/D4.
