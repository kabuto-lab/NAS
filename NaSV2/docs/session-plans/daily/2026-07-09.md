pub struct HmacJwtIssuer {
    encoding_key: EncodingKey,
}

impl HmacJwtIssuer {
    #[must_use]
    pub fn new(secret: &SecretString) -> Self {
        Self {
            encoding_key: EncodingKey::from_secret(secret.expose_secret().as_bytes()),
        }
    }
}

#[async_trait]
impl JwtIssuer for HmacJwtIssuer {
    async fn issue(&self, claims: JwtClaims) -> Result<String, AppError> {
        encode(&Header::new(Algorithm::HS256), &claims, &self.encoding_key)
            .map_err(|e| { tracing::error!(error=%e); AppError::Internal })
    }
}

P2 · `crates/application/src/commands/mod.rs` + `login.rs` (NEW)

```rust
//! Login command — assembles user lookup + password verify +
//! capability resolution + token issuance.

use std::sync::Arc;

use chrono::Utc;
use nas2_common::{AppError, TenantId, UserId};
use nas2_domain::{Capability, CapabilitySet, Email};
use uuid::Uuid;

use crate::auth::password::verify_password;
use crate::auth::jwt::{JwtClaims, JwtIssuer};

/// What the handler returns — token + minimal user info for client
/// UX (no PII beyond email).
#[derive(Debug, Clone)]
pub struct LoginResult {
    pub token: String,
    pub user_id: UserId,
    pub caps: Vec<String>,
}

/// Trait the command depends on for password+caps lookup. Implemented
/// in infrastructure (PgUserRepository::find_by_email_with_hash plus
/// a separate `caps_for_user(tenant, user_id)` query joining
/// role_capabilities).
#[async_trait::async_trait]
pub trait LoginUserLookup: Send + Sync {
    async fn find_with_hash(
        &self,
        tenant: TenantId,
        email: &Email,
    ) -> Result<Option<(UserId, String /* hash */)>, AppError>;

    async fn caps_for(
        &self,
        tenant: TenantId,
        user_id: UserId,
    ) -> Result<CapabilitySet, AppError>;
}

pub struct Login<L: LoginUserLookup, I: JwtIssuer> {
    pub lookup: L,
    pub issuer: I,
    pub issuer_name: String,
    pub token_ttl_secs: i64,
}

impl<L: LoginUserLookup, I: JwtIssuer> Login<L, I> {
    pub async fn execute(
        &self,
        tenant: TenantId,
        email: &Email,
        plain: &str,
    ) -> Result<LoginResult, AppError> {
        let Some((user_id, hash)) = self.lookup.find_with_hash(tenant, email).await? else {
            // Same error for missing user or wrong password — denies
            // enumeration via response-time / status diff.
            return Err(AppError::Forbidden("invalid_credentials".into()));
        };
        if !verify_password(plain, &hash)? {
            return Err(AppError::Forbidden("invalid_credentials".into()));
        }
        let caps = self.lookup.caps_for(tenant, user_id).await?;
        let now = Utc::now().timestamp();
        let claims = JwtClaims {
            sub: user_id,
            tenant,
            caps: caps.iter().map(|c| c.as_str().to_owned()).collect(),
            iat: now,
            exp: now + self.token_ttl_secs,
            jti: Uuid::new_v4(),
            iss: self.issuer_name.clone(),
        };
        let token = self.issuer.issue(claims.clone()).await?;
        Ok(LoginResult { token, user_id, caps: claims.caps })
    }
}
```

P3 · Implement `LoginUserLookup` on top of `PgUserRepository`

  In `crates/infrastructure/src/persistence/user.rs`, add at the
  bottom:

```rust
#[async_trait::async_trait]
impl nas2_application::commands::login::LoginUserLookup for PgUserRepository {
    async fn find_with_hash(
        &self,
        tenant: TenantId,
        email: &Email,
    ) -> Result<Option<(UserId, String)>, AppError> {
        Ok(self.find_by_email_with_hash(tenant, email).await?
            .map(|(u, h)| (u.id, h)))
    }

    async fn caps_for(
        &self,
        tenant: TenantId,
        user_id: UserId,
    ) -> Result<CapabilitySet, AppError> {
        let uid = user_id.into_uuid();
        with_tenant(&self.pool, tenant, |tx| async move {
            let rows: Vec<(String,)> = sqlx::query_as(
                "SELECT DISTINCT rc.capability \
                 FROM role_capabilities rc \
                 JOIN user_role_assignments ura ON ura.role_id = rc.role_id \
                 WHERE ura.user_id = $1"
            ).bind(uid).fetch_all(&mut **tx).await
             .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
            let mut set = CapabilitySet::new();
            for (cap_str,) in rows {
                if let Ok(cap) = Capability::try_from_str(&cap_str) {
                    set.insert(cap);
                }
                // Unknown caps in DB are silently dropped (deploy lag
                // tolerance); admin tools should reject unknown caps
                // at write boundary in M3+.
            }
            Ok::<_, AppError>(set)
        }).await
    }
}
```

```rust
//! Auth endpoints — login (today). password-reset etc. land in M3+.

use axum::{
    extract::{Extension, State},
    http::{header, HeaderMap, HeaderValue, StatusCode},
    response::IntoResponse,
    Json,
};
use serde::{Deserialize, Serialize};

use nas2_application::auth::jwt::JwtIssuer;
use nas2_application::commands::login::{Login, LoginUserLookup};
use nas2_common::AppError;
use nas2_domain::Email;
use nas2_tenant::TenantContext;

use crate::app_state::AuthAppState; // see P5 wiring

#[derive(Debug, Deserialize)]
pub struct LoginBody {
    email: String,
    password: String,
}

#[derive(Debug, Serialize)]
pub struct LoginResponse {
    user_id: String,
    caps: Vec<String>,
    // token in body too, so non-cookie clients can use it
    token: String,
}

/// no_capability_required: public auth endpoint
pub async fn login(
    State(state): State<AuthAppState>,
    Extension(tctx): Extension<TenantContext>,
    Json(body): Json<LoginBody>,
) -> Result<impl IntoResponse, AppError> {
    let email = Email::try_new(&body.email)?;
    let cmd = Login {
        lookup: state.user_repo.clone(),
        issuer: state.jwt_issuer.clone(),
        issuer_name: state.jwt_issuer_name.clone(),
        token_ttl_secs: state.jwt_ttl_secs,
    };
    let result = cmd.execute(tctx.tenant_id, &email, &body.password).await?;

    let cookie = format!(
        "ax_session={token}; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age={maxage}",
        token = result.token,
        maxage = state.jwt_ttl_secs,
    );
    let mut headers = HeaderMap::new();
    headers.insert(header::SET_COOKIE,
                   HeaderValue::from_str(&cookie)
                       .map_err(|_| AppError::Internal)?);

    let body = LoginResponse {
        user_id: result.user_id.to_string(),
        caps: result.caps,
        token: result.token,
    };
    Ok((StatusCode::OK, headers, Json(body)))
}
```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Same error for "no user" and "wrong password"** — critical
     for anti-enumeration. Don't be tempted to add a more helpful
     "user not found" message.
  2. **Cookie attributes**: Secure requires HTTPS. In dev (HTTP),
     Secure causes the browser to drop the cookie. Use an env-var
     to toggle Secure on (default on, off for dev).
  3. **caps_for query** joins `role_capabilities` + `user_role_assignments`
     — both have RLS policies (0007). The `with_tenant` block sets
     the GUC; RLS filters correctly. Verify via test.
  4. **JwtIssuer + JwtVerifier symmetric key** — share via the same
     SecretString in apps/server boot. Single key for now (M3 rotation).
  5. **tower-governor not in workspace deps** is possible — if so,
     document a follow-up; ship without rate limit; M3 boot adds it.
