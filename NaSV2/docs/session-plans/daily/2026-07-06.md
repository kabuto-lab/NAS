```rust
//! JWT verification port.
//!
//! Token shape (claims):
//!   sub      — UserId (UUID)
//!   tenant   — TenantId (UUID)
//!   caps     — Vec<String> (capability stable names per
//!              `Capability::as_str()`)
//!   iat      — issued-at unix-seconds
//!   exp      — expiration unix-seconds
//!   jti      — JWT id (UUID, for replay detection)
//!   iss      — issuer (constant string per deployment)

use async_trait::async_trait;
use nas2_common::{AppError, TenantId, UserId};
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct JwtClaims {
    pub sub: UserId,
    pub tenant: TenantId,
    pub caps: Vec<String>,
    pub iat: i64,
    pub exp: i64,
    pub jti: Uuid,
    pub iss: String,
}

#[async_trait]
pub trait JwtVerifier: Send + Sync {
    /// Verify a Bearer token (or cookie value).
    ///
    /// Errors:
    ///   - `AppError::Forbidden("expired_token")` — exp < now
    ///   - `AppError::Forbidden("invalid_signature")` — HMAC mismatch
    ///   - `AppError::Forbidden("future_iat")` — iat > now + skew
    ///   - `AppError::Forbidden("malformed_token")` — JWT parse fail
    ///   - `AppError::Forbidden("wrong_issuer")` — iss mismatch
    ///   - `AppError::Forbidden("replay_detected")` — jti seen in window
    async fn verify(&self, token: &str) -> Result<JwtClaims, AppError>;
}
```

`jwt.rs`:

```rust
use async_trait::async_trait;
use jsonwebtoken::{decode, errors::ErrorKind, Algorithm, DecodingKey, Validation};
use nas2_application::auth::{JwtClaims, JwtVerifier};
use nas2_common::AppError;
use secrecy::{ExposeSecret, SecretString};

pub struct HmacJwtVerifier {
    decoding_key: DecodingKey,
    expected_issuer: String,
    // Clock skew tolerance in seconds — 60 default
    leeway_seconds: u64,
}

impl HmacJwtVerifier {
    #[must_use]
    pub fn new(secret: &SecretString, expected_issuer: impl Into<String>) -> Self {
        Self {
            decoding_key: DecodingKey::from_secret(secret.expose_secret().as_bytes()),
            expected_issuer: expected_issuer.into(),
            leeway_seconds: 60,
        }
    }

    #[must_use]
    pub const fn with_leeway(mut self, sec: u64) -> Self {
        self.leeway_seconds = sec;
        self
    }
}

#[async_trait]
impl JwtVerifier for HmacJwtVerifier {
    async fn verify(&self, token: &str) -> Result<JwtClaims, AppError> {
        let mut v = Validation::new(Algorithm::HS256);
        v.set_issuer(&[&self.expected_issuer]);
        v.leeway = self.leeway_seconds;
        v.required_spec_claims = ["exp", "iat", "iss", "sub"].iter().map(|s| (*s).to_owned()).collect();

        match decode::<JwtClaims>(token, &self.decoding_key, &v) {
            Ok(data) => {
                // jsonwebtoken validates exp, iat (against future), iss.
                // Additional client check: ensure jti is a real UUID
                // (already enforced by serde).
                Ok(data.claims)
            }
            Err(e) => Err(map_jwt_err(e)),
        }
    }
}

fn map_jwt_err(e: jsonwebtoken::errors::Error) -> AppError {
    let code = match e.kind() {
        ErrorKind::ExpiredSignature   => "expired_token",
        ErrorKind::InvalidSignature   => "invalid_signature",
        ErrorKind::ImmatureSignature  => "future_iat",
        ErrorKind::InvalidIssuer      => "wrong_issuer",
        ErrorKind::InvalidToken
        | ErrorKind::InvalidAlgorithm
        | ErrorKind::Base64(_)
        | ErrorKind::Json(_)
        | ErrorKind::Utf8(_)          => "malformed_token",
        _                              => "malformed_token",
    };
    AppError::Forbidden(code.to_owned())
}
```

  ```rust
  #[cfg(test)]
  #[allow(clippy::expect_used, clippy::unwrap_used, clippy::panic)]
  mod tests {
      use super::*;
      use jsonwebtoken::{encode, EncodingKey, Header};
      use nas2_common::{TenantId, UserId};
      use secrecy::SecretString;
      use uuid::Uuid;

      fn sign(claims: &JwtClaims, key: &[u8]) -> String {
          encode(&Header::new(Algorithm::HS256), claims,
                 &EncodingKey::from_secret(key)).unwrap()
      }

      fn valid_claims(skew_secs: i64) -> JwtClaims {
          let now = chrono::Utc::now().timestamp();
          JwtClaims {
              sub: UserId::new_v4(),
              tenant: TenantId::new_v4(),
              caps: vec!["cms.page.read".to_owned()],
              iat: now + skew_secs,
              exp: now + 3600,
              jti: Uuid::new_v4(),
              iss: "ax-cms".to_owned(),
          }
      }

      #[tokio::test]
      async fn verifies_valid_token() {
          let key = b"k".repeat(32);
          let claims = valid_claims(0);
          let tok = sign(&claims, &key);
          let v = HmacJwtVerifier::new(
              &SecretString::from(String::from_utf8(key.clone()).unwrap()),
              "ax-cms");
          let got = v.verify(&tok).await.unwrap();
          assert_eq!(got.sub, claims.sub);
      }

      #[tokio::test]
      async fn rejects_expired_token() {
          let key = b"k".repeat(32);
          let mut c = valid_claims(0);
          c.exp = chrono::Utc::now().timestamp() - 10;
          let tok = sign(&c, &key);
          let v = HmacJwtVerifier::new(
              &SecretString::from(String::from_utf8(key).unwrap()),
              "ax-cms");
          let err = v.verify(&tok).await.unwrap_err();
          assert!(matches!(err, AppError::Forbidden(s) if s == "expired_token"));
      }

      #[tokio::test]
      async fn rejects_bad_signature() {
          let key_signing = b"k1".repeat(16);
          let key_verifying = b"k2".repeat(16);
          let c = valid_claims(0);
          let tok = sign(&c, &key_signing);
          let v = HmacJwtVerifier::new(
              &SecretString::from(String::from_utf8(key_verifying).unwrap()),
              "ax-cms");
          let err = v.verify(&tok).await.unwrap_err();
          assert!(matches!(err, AppError::Forbidden(s) if s == "invalid_signature"));
      }
  ```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`jsonwebtoken::Validation` defaults validate exp + nbf — NOT
     iss.** Must call `set_issuer(&[&iss])` explicitly.
  2. **HMAC key length.** HS256 minimum 256 bits (32 bytes). Tests
     use `b"k".repeat(32)` for 32 bytes — meets minimum. Document in
     RFC-006 that env-loaded secret must be ≥ 32 bytes.
  3. **`SecretString` is from `secrecy` crate** — auto-zeroes on
     drop; `expose_secret()` returns `&str`. Don't clone the inner.
  4. **Algorithm flexibility**: only HS256 supported today. RFC-006
     notes RS256/ES256 as month-3 ADR if external IDPs need it.
  5. **`InvalidAlgorithm` shouldn't appear** in production — but if
     a token with `alg=none` arrives, our mapping returns
     `malformed_token` (correct).
