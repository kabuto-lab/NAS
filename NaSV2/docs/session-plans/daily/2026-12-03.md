```rust
// list_users.rs
pub struct ListUsers<U: UserRepository> { repo: U }

impl<U: UserRepository> ListUsers<U> {
    pub async fn execute(
        &self, tenant: TenantId, page: u32, per_page: u32, caps: &CapabilitySet,
    ) -> Result<Page<UserSummary>, AppError> {
        if !caps.contains(Capability::ManageUsers) { return Err(...); }
        self.repo.list_with_role_count(tenant, page, per_page).await
    }
}

#[derive(Clone, Debug)]
pub struct UserSummary {
    pub id: UserId,
    pub email: Email,
    pub role_count: usize,
    pub created_at: DateTime<Utc>,
}
```

```rust
// get_user_with_roles.rs
pub struct GetUserWithRoles<U: UserRepository, R: RoleRepository> {
    users: U, roles: R,
}

#[derive(Clone, Debug)]
pub struct UserWithRoles {
    pub user: User,
    pub roles: Vec<Role>,
    pub combined_caps: CapabilitySet,
}

impl<U: UserRepository, R: RoleRepository> GetUserWithRoles<U, R> {
    pub async fn execute(
        &self, tenant: TenantId, user_id: UserId, caps: &CapabilitySet,
    ) -> Result<UserWithRoles, AppError> {
        if !caps.contains(Capability::ManageUsers) { return Err(...); }
        let user = self.users.find_by_id(tenant, user_id).await?
            .ok_or(AppError::NotFound(...))?;
        let roles = self.roles.find_by_ids(tenant, &user.role_ids).await?;
        let combined_caps = roles.iter()
            .flat_map(|r| r.capabilities.iter())
            .collect::<CapabilitySet>();
        Ok(UserWithRoles { user, roles, combined_caps })
    }
}
```

```rust
#[component]
pub fn UsersList(initial: Page<UserSummary>) -> impl IntoView {
    view! {

#[component]
pub fn UserDetail(data: UserWithRoles) -> impl IntoView {
    let all_caps: Vec<Capability> = vec![
        Capability::CmsPageRead, Capability::CmsPageDraft, Capability::CmsPagePublish,
        Capability::MediaUpload, Capability::ManageUsers, Capability::ManageSite,
        Capability::CommentsModerate,
    ];
    let roles = data.roles.clone();
    view! {
        <section class="user-detail">
            <header>
                <h2>{data.user.email.as_str().to_owned()}</h2>
                <span class="badge muted">"Created " {data.user.created_at.format("%Y-%m-%d").to_string()}</span>
            </header>
            <h3>"Assigned roles"</h3>
            <ul class="role-list">
                {data.roles.iter().map(|r| view! {
                    <li><span class="role-chip">{r.name.clone()}</span></li>
                }).collect_view()}
            </ul>
            <h3>"Capability matrix"</h3>
            <table class="cap-matrix">
                <thead>
                    <tr>
                        <th>"Capability"</th>
                        {roles.iter().map(|r| view! { <th>{r.name.clone()}</th> }).collect_view()}
                        <th>"Effective"</th>
                    </tr>
                </thead>
                <tbody>
                    {all_caps.into_iter().map(|cap| {
                        let cap_str = cap.as_str();
                        let effective = data.combined_caps.contains(cap);
                        let roles_clone = roles.clone();
                        view! {
                            <tr>
                                <td><code>{cap_str}</code></td>
                                {roles_clone.iter().map(|r| {
                                    let has = r.capabilities.contains(cap);
                                    view! { <td class:has=has class:no=!has>{if has {"✓"} else {""}}</td> }
                                }).collect_view()}
                                <td class:effective=effective>{if effective {"✓"} else {"—"}}</td>
                            </tr>
                        }
                    }).collect_view()}
                </tbody>
            </table>
            <p class="callout muted">
                "Role assignment / revocation UI lands in M9 (extension API
                + admin polish). Use nas2-cli for now."
            </p>
        </section>
    }
}
```

```rust
// caps.require(manage.users)
pub async fn users_list(
    _: RequireCaps<ManageUsers>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Query(q): Query<PaginationQuery>,
) -> Result<impl IntoResponse, AppError> {
    let uc = ListUsers::new(state.user_repo.clone());
    let page = uc.execute(tctx.tenant_id, q.page, q.per_page, &caps).await?;
    let ctx = build_admin_ctx(...);
    let crumbs = BreadcrumbChain::root().push(Crumb::terminal("Users"));
    let html = render_to_string(move || view! {
        <AdminLayout ctx title="Users" crumbs>
            <UsersList initial=page/>
        </AdminLayout>
    });
    Ok(Html(html.into_owned()))
}

// caps.require(manage.users)
pub async fn user_detail(
    _: RequireCaps<ManageUsers>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Path(id_raw): Path<String>,
) -> Result<impl IntoResponse, AppError> {
    let id = UserId::from_uuid(uuid::Uuid::parse_str(&id_raw)
        .map_err(|_| AppError::Validation("invalid user id".into()))?);
    let uc = GetUserWithRoles::new(state.user_repo.clone(), state.role_repo.clone());
    let data = uc.execute(tctx.tenant_id, id, &caps).await?;
    let ctx = build_admin_ctx(...);
    let crumbs = BreadcrumbChain::root()
        .push(Crumb::linked("Users", "/admin/users"))
        .push(Crumb::terminal(data.user.email.as_str().to_owned()));
    let html = render_to_string(move || view! {
        <AdminLayout ctx title="User detail" crumbs>
            <UserDetail data/>
        </AdminLayout>
    });
    Ok(Html(html.into_owned()))
}
```
