P2 · Real-DB integration test for the handler

  Append to `crates/presentation/tests/get_page_by_slug.rs`:

```rust
#[tokio::test]
#[ignore = "needs Docker"]
async fn returns_200_with_real_pg_repo() {
    // Boot PG + migrations + repo
    let pg = Postgres::default()
        .with_user("postgres").with_password("postgres").with_db_name("ax")
        .start().await.unwrap();
    let port = pg.get_host_port_ipv4(5432).await.unwrap();
    let url = format!("postgres://postgres:postgres@127.0.0.1:{port}/ax");
    let pool = PgPoolOptions::new().connect(&url).await.unwrap();
    for n in 2..=8 {
        if let Some(p) = glob::glob(&format!("../../migrations/{n:04}_*.sql"))
            .unwrap().filter_map(Result::ok).next() {
            let sql = std::fs::read_to_string(&p).unwrap();
            sqlx::query(&sql).execute(&pool).await.unwrap();
        }
    }

    let repo = nas2_infrastructure::PgPostRepository::new(pool.clone());

    // Seed: tenant, site, post; resolver map
    let tenant = nas2_common::TenantId::new_v4();
    let site = nas2_common::SiteId::new_v4();
    // ... seed via with_tenant (as in W6 D2 helper) ...
    seed_via_repo(&pool, &repo, tenant, site, "hello").await;

    // Build resolver mapping host → (tenant, site)
    let mut m = HashMap::new();
    m.insert("example.com".to_owned(),
             nas2_tenant::TenantContext::new(tenant, site));
    let resolver = std::sync::Arc::new(InMemoryTenantResolver::new(m));

    // Build full app
    let state = AppState::new(std::sync::Arc::new(repo), resolver);
    let app = build_router(state);

    let req = Request::builder()
        .uri("/api/v1/pages/hello")
        .header(header::HOST, "example.com")
        .body(Body::empty()).unwrap();
    let resp = app.oneshot(req).await.unwrap();
    assert_eq!(resp.status(), StatusCode::OK);
}
```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Spine touch on apps/server/main.rs.** Authorised — keep the
     diff minimal (<= 10 lines). Verify with `git diff apps/server/`.
  2. **`AppState::new` signature change** — existing tests using
     MockRepo still compile because the type is `Arc<dyn
     PostRepository>`. If you accidentally narrow to `Arc<PgPostRepository>`,
     breaks tests.
  3. **`http_pool` vs `worker_pool` for PgPostRepository.** Use
     `http_pool` (request-path queries are PostRepository methods).
     Worker pool will get its own repo holdings in M3.
  4. **`build_tenant_resolver` env-var format**. Pick something
     dev-friendly and trivial — CSV `host=tenant,site` per ENTITY
     §15 dev seed.
  5. **`#[ignore]` integration test path globs `../../migrations/`.**
     Run via `cargo test -p nas2-presentation --tests -- --ignored`
     from the workspace root; `cargo test` runs in `crates/presentation/`
     so the relative path is `../../migrations/`.
