```rust
#[component]
pub fn BlockNode(index: usize, block: Block) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let is_editing = Memo::new(move |_| state.editing_index.get() == Some(index));
    let enter_edit = move |_| state.editing_index.set(Some(index));
    let on_keydown = move |ev: web_sys::KeyboardEvent| {
        if ev.ctrl_key() || ev.meta_key() {
            match ev.key().as_str() {
                "ArrowUp"   => { ev.prevent_default(); state.move_up(index); }
                "ArrowDown" => { ev.prevent_default(); state.move_down(index); }
                _ => {}
            }
        }
        if ev.key() == "Escape" && is_editing.get_untracked() {
            state.editing_index.set(None);
        }
    };

    view! {
        <article class="block-node"
                 tabindex="0"
                 class:editing=is_editing
                 on:click=enter_edit
                 on:keydown=on_keydown>
            <aside class="block-toolbar">/* unchanged from D2/D3 */</aside>
            <div class="block-body">
                <Show when=move || is_editing.get()
                      fallback=move || view! { <BlockViewByVariant index block/> }>
                    <BlockEditByVariant index/>
                </Show>
            </div>
        </article>
    }
}
```

```rust
pub fn BlockEditByVariant(index: usize) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let block = Memo::new(move |_| state.blocks.with(|bs| bs.get(index).cloned()));
    view! {
        {move || match block.get() {
            Some(Block::Heading { .. })  => view! { <heading::HeadingEdit index/> }.into_any(),
            Some(Block::Paragraph { .. })=> view! { <paragraph::ParagraphEdit index/> }.into_any(),
            Some(Block::Image { .. })    => view! { <image::ImageEdit index/> }.into_any(),
            Some(Block::CodeBlock { .. })=> view! { <code_block::CodeBlockEdit index/> }.into_any(),
            _                            => view! { <placeholder::PlaceholderEdit/> }.into_any(),
        }}
    }
}
```

P3 · `editor/variants/heading.rs::HeadingEdit`

```rust
use nas2_domain::{Block, HeadingLevel};

#[component]
pub fn HeadingEdit(index: usize) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let (text_get, level_get) = compute_initial(state.clone(), index);

    let update_text = move |ev| {
        let new_text = event_target_value(&ev);
        state.blocks.update(|bs| {
            if let Some(Block::Heading { text, .. }) = bs.get_mut(index) {
                *text = new_text;
            }
        });
    };
    let update_level = move |ev| {
        let level_str = event_target_value(&ev);
        let new_level = parse_level(&level_str);
        state.blocks.update(|bs| {
            if let Some(Block::Heading { level, .. }) = bs.get_mut(index) {
                *level = new_level;
            }
        });
    };
    view! {
        <div class="edit-heading">
            <select on:change=update_level>
                {(1..=6).map(|n| {
                    let v = format!("h{n}");
                    let sel = level_get.get() == HeadingLevel::from_u8(n);
                    view! { <option value=v.clone() selected=sel>{format!("H{n}")}</option> }
                }).collect_view()}
            </select>
            <input type="text" placeholder="Heading text"
                   prop:value=move || text_get.get()
                   on:input=update_text/>
        </div>
    }
}

fn compute_initial(state: EditorState, index: usize) -> (Memo<String>, Memo<HeadingLevel>) {
    let text = Memo::new(move |_| state.blocks.with(|bs| {
        if let Some(Block::Heading { text, .. }) = bs.get(index) { text.clone() } else { String::new() }
    }));
    let level = Memo::new(move |_| state.blocks.with(|bs| {
        if let Some(Block::Heading { level, .. }) = bs.get(index) { *level } else { HeadingLevel::H2 }
    }));
    (text, level)
}

fn parse_level(s: &str) -> HeadingLevel {
    match s {
        "h1" => HeadingLevel::H1, "h2" => HeadingLevel::H2, "h3" => HeadingLevel::H3,
        "h4" => HeadingLevel::H4, "h5" => HeadingLevel::H5, "h6" => HeadingLevel::H6,
        _ => HeadingLevel::H2,
    }
}
```

```rust
#[component]
pub fn ParagraphEdit(index: usize) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let html = Memo::new(move |_| state.blocks.with(|bs| {
        if let Some(Block::Paragraph { html }) = bs.get(index) { html.clone() } else { String::new() }
    }));
    let update = move |ev| {
        let new_html = event_target_value(&ev);
        state.blocks.update(|bs| {
            if let Some(Block::Paragraph { html }) = bs.get_mut(index) { *html = new_html; }
        });
    };
    view! {
        <div class="edit-paragraph">
            <textarea rows=4 placeholder="Paragraph text (basic HTML supported)"
                      prop:value=move || html.get()
                      on:input=update/>
            <p class="hint muted">"Tip: <strong>, <em>, <a> are allowed. Other tags stripped on save."</p>
        </div>
    }
}
```

PERFORMANCE / SECURITY / PITFALLS
═════════════════════════════════════════════════════════════════════

  - Edit-mode signal updates flow on every keystroke — fine for
    small posts; debounce considered but rejected for MVP (state is
    in-memory, no network).
  - ParagraphEdit `<textarea>` accepts raw HTML; sanitization runs
    at SAVE time (W3 D5 + M2 W8 D4 ammonia at PostRepository::update).
    Today's preview renders sanitised version of saved state, NOT
    live preview.
  - ImageEdit URL input accepts any string — Block::Image's
    `src: String` is a placeholder until M3 Media model swap to
    MediaId. Pasted URLs must be validated server-side before save.
  - Escape handler reads `is_editing.get_untracked()` to avoid an
    infinite reactive loop.
  - Click-bubbling: clicking on the textarea inside an editing block
    should NOT propagate to the article's on:click (which would
    re-trigger enter_edit — harmless but noisy). Use `stop_propagation`
    on inner form elements.
