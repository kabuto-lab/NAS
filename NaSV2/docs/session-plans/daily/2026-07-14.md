```rust
//! Update a page in-place. Partial PATCH semantics: Option<T> per
//! field; None = unchanged.

use chrono::Utc;
use nas2_common::{AppError, SiteId, TenantId};
use nas2_domain::{Block, Capability, CapabilitySet, Post, PostSlug, PostStatus};
use crate::ports::PostRepository;

#[derive(Debug, Clone, Default)]
pub struct UpdatePageInput {
    pub title: Option<String>,
    pub slug: Option<String>,
    pub blocks: Option<Vec<Block>>,
    pub status: Option<PostStatus>,
}

pub struct UpdatePage<R> { repo: R }
impl<R: PostRepository> UpdatePage<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        site: SiteId,
        slug: &PostSlug,
        input: UpdatePageInput,
        caps: &CapabilitySet,
    ) -> Result<Post, AppError> {
        // Caps gate — split by intent.
        let touches_status_publish_path = matches!(input.status,
            Some(PostStatus::Published) | Some(PostStatus::Scheduled));
        if touches_status_publish_path {
            if !caps.contains(Capability::CmsPagePublish) {
                return Err(AppError::Forbidden(Capability::CmsPagePublish.as_str().to_owned()));
            }
        } else if !caps.contains(Capability::CmsPageDraft) {
            return Err(AppError::Forbidden(Capability::CmsPageDraft.as_str().to_owned()));
        }

        let Some(mut post) = self.repo.find_by_slug(tenant, site, slug).await? else {
            return Err(AppError::NotFound(format!("page slug={}", slug.as_str())));
        };

        // Apply patch field-by-field.
        if let Some(t) = input.title {
            if t.trim().is_empty() {
                return Err(AppError::Validation("title cannot be blank".into()));
            }
            post.title = t;
        }
        if let Some(s) = input.slug {
            let new_slug = PostSlug::try_new(&s)?;
            post.slug = new_slug;
        }
        if let Some(bs) = input.blocks {
            // TODO M4: Block::validate_each(&bs)?
            post.blocks = bs;
        }
        if let Some(new_status) = input.status {
            post.transition_to(new_status, Utc::now())?;
        }

        self.repo.update(&post).await?;
        Ok(post)
    }
}
```

```rust
#[derive(Debug, Deserialize)]
pub struct UpdatePageBody {
    pub title: Option<String>,
    pub slug: Option<String>,
    pub blocks: Option<Vec<nas2_domain::Block>>,
    pub status: Option<nas2_domain::PostStatus>,
}

/// caps.require(cms.page.draft) — title/blocks
/// caps.require(cms.page.publish) — status → Published/Scheduled
pub async fn update_page(
    State(state): State<AppState>,
    Extension(tctx): Extension<TenantContext>,
    Extension(caps): Extension<CapabilitySet>,
    Path(slug_raw): Path<String>,
    Json(body): Json<UpdatePageBody>,
) -> Result<Json<Post>, AppError> {
    use nas2_application::commands::update_page::{UpdatePage, UpdatePageInput};
    let slug = PostSlug::try_new(&slug_raw)?;
    let cmd = UpdatePage::new(state.post_repo.clone());
    let post = cmd.execute(
        tctx.tenant_id, tctx.site_id, &slug,
        UpdatePageInput {
            title: body.title, slug: body.slug,
            blocks: body.blocks, status: body.status,
        },
        &caps,
    ).await?;
    Ok(Json(post))
}
```
