```rust
//! Hydrated bulk-select island for PostList · M7 W2 D4.
//!
//! Per ADR-019: signal-based + provide_context for cross-component
//! state. This is the first M7 island that needs CLIENT-SIDE
//! reactivity (selection count updates).

use leptos::prelude::*;
use nas2_common::PostId;

use crate::admin::context::AdminContext;

#[derive(Clone, Debug)]
pub struct BulkSelection {
    pub selected: RwSignal<Vec<PostId>>,
    pub all_ids: Vec<PostId>,
}

impl BulkSelection {
    pub fn new(all_ids: Vec<PostId>) -> Self {
        Self { selected: RwSignal::new(Vec::new()), all_ids }
    }
}

#[component]
pub fn BulkSelectProvider(all_ids: Vec<PostId>, children: Children) -> impl IntoView {
    let selection = BulkSelection::new(all_ids);
    provide_context(selection);
    view! { {children()} }
}

#[component]
pub fn RowCheckbox(id: PostId) -> impl IntoView {
    let selection = expect_context::<BulkSelection>();
    let checked = Memo::new(move |_| selection.selected.get().contains(&id));
    let toggle = move |_| {
        selection.selected.update(|s| {
            if let Some(pos) = s.iter().position(|x| *x == id) {
                s.remove(pos);
            } else {
                s.push(id);
            }
        });
    };
    view! {
        <input type="checkbox" name="ids[]" value=id.to_string()
               prop:checked=checked on:change=toggle/>
    }
}

#[component]
pub fn HeaderCheckbox() -> impl IntoView {
    let selection = expect_context::<BulkSelection>();
    let all_selected = Memo::new(move |_| {
        let sel = selection.selected.get();
        !selection.all_ids.is_empty() && sel.len() == selection.all_ids.len()
    });
    let toggle_all = move |_| {
        selection.selected.update(|s| {
            if !s.is_empty() && s.len() == selection.all_ids.len() {
                s.clear();
            } else {
                *s = selection.all_ids.clone();
            }
        });
    };
    view! {
        <input type="checkbox" prop:checked=all_selected
               on:change=toggle_all aria-label="select all"/>
    }
}

#[component]
pub fn BulkActionBar() -> impl IntoView {
    let selection = expect_context::<BulkSelection>();
    let ctx = expect_context::<AdminContext>();
    let count = Memo::new(move |_| selection.selected.get().len());
    let visible = move || count.get() > 0;
    view! {
        <Show when=visible fallback=|| ()>
            <form method="post" action="/admin/posts/bulk" class="bulk-bar">
                <span class="bulk-count">{move || format!("{} selected", count.get())}</span>
                <input type="hidden" name="csrf" value=ctx.csrf_token.clone()/>
                {move || {
                    selection.selected.get().into_iter().map(|id| view! {
                        <input type="hidden" name="ids[]" value=id.to_string()/>
                    }).collect_view()
                }}
                <select name="action">
                    <option value="archive">"Archive"</option>
                    <option value="set_draft">"Set Draft"</option>
                    <option value="set_published">"Set Published"</option>
                </select>
                <button type="submit" class="btn">"Apply"</button>
            </form>
        </Show>
    }
}
```

P4 · Client-side hydration glue

  Leptos 0.7 requires the admin pages to opt-in to hydration. The
  existing AdminLayout uses pure SSR. To enable an island, the
  presentation crate's `lib.rs` must export the WASM bundle build
  config — but for M7 we use **isomorphic hydration** without WASM
  by leveraging Leptos's `hydrate_islands` mode (Leptos 0.7 feature).

  If `hydrate_islands` is not yet wired in this codebase (likely),
  today's island gracefully degrades to non-interactive:
  the SSR rendering is fully functional via GET form submits + the
  checkbox state persists across reload (browser caches checkbox
  state in many cases).

  Decision today: **ship SSR-only for D4**. The bulk bar HTML is
  always rendered (no `Show when` JS dependency); we use a `<details>`
  element trick:

```rust
// Simplified non-hydrated version
view! {
    <details class="bulk-details" open=false>
        <summary class="bulk-summary">"Bulk actions ▾"</summary>
        <form method="post" action="/admin/posts/bulk" class="bulk-bar">
            // ... form content ...
        </form>
    </details>
}
```

  HTML `<details>` provides reveal-on-click natively without JS.
  Selection count is updated by re-rendering after POST (operator
  picks rows, clicks Apply, gets new page).

  **AI-Default declared:** ship `<details>`-based bulk UI today;
  full reactive Leptos-hydrated island deferred to a `M7.5`
  catch-up or M8 polish (record in SESSION_LOG). Tests update
  accordingly.
