```rust
//! Create a draft page in a site.
//!
//! Capability: `cms.page.draft` (creator does not need to be able
//! to publish — that's `cms.page.publish` and goes through
//! UpdatePage::transition_to).

use chrono::Utc;
use nas2_common::{AppError, PostId, SiteId, TenantId};
use nas2_domain::{Block, Capability, CapabilitySet, Post, PostSlug, PostStatus};

use crate::ports::PostRepository;

pub struct CreatePage<R> { repo: R }

#[derive(Debug, Clone)]
pub struct CreatePageInput {
    pub site_id: SiteId,
    pub slug: String,           // validated inside execute
    pub title: String,
    pub blocks: Vec<Block>,
    pub custom_type: Option<String>,
}

impl<R: PostRepository> CreatePage<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        input: CreatePageInput,
        caps: &CapabilitySet,
    ) -> Result<Post, AppError> {
        if !caps.contains(Capability::CmsPageDraft) {
            return Err(AppError::Forbidden(Capability::CmsPageDraft.as_str().to_owned()));
        }
        let slug = PostSlug::try_new(&input.slug)?;
        if input.title.trim().is_empty() {
            return Err(AppError::Validation("title must be non-empty".into()));
        }
        let post = Post {
            id: PostId::new_v4(),
            site_id: input.site_id,
            tenant_id: tenant,
            slug,
            title: input.title,
            status: PostStatus::Draft,
            published_at: None,
            blocks: input.blocks,
            custom_type: input.custom_type.map(nas2_domain::CustomPostType),
        };
        // TODO future: Block::validate_each(&post.blocks)? — M4
        self.repo.insert(&post).await?;
        Ok(post)
    }
}
```

P2 · Handler

```rust
use serde::{Deserialize, Serialize};

#[derive(Debug, Deserialize)]
pub struct CreatePageBody {
    pub site_id: uuid::Uuid,
    pub slug: String,
    pub title: String,
    #[serde(default)]
    pub blocks: Vec<nas2_domain::Block>,
    pub custom_type: Option<String>,
}

#[derive(Debug, Serialize)]
pub struct CreatedPage { pub id: String, pub slug: String }

/// caps.require(cms.page.draft)
pub async fn create_page(
    State(state): State<AppState>,
    Extension(tctx): Extension<TenantContext>,
    Extension(caps): Extension<CapabilitySet>,
    Json(body): Json<CreatePageBody>,
) -> Result<(StatusCode, Json<CreatedPage>), AppError> {
    use nas2_application::commands::create_page::{CreatePage, CreatePageInput};
    let cmd = CreatePage::new(state.post_repo.clone());
    let post = cmd.execute(
        tctx.tenant_id,
        CreatePageInput {
            site_id: SiteId::from_uuid(body.site_id),
            slug: body.slug,
            title: body.title,
            blocks: body.blocks,
            custom_type: body.custom_type,
        },
        &caps,
    ).await?;
    Ok((StatusCode::CREATED, Json(CreatedPage {
        id: post.id.to_string(),
        slug: post.slug.as_str().to_owned(),
    })))
}
```
