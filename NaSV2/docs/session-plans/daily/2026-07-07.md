```rust
//! JWT authentication middleware.
//!
//! Extracts token from (a) Authorization: Bearer <token>, (b)
//! Cookie: ax_session=<token>. Verifies via `JwtVerifier`,
//! enforces replay protection via moka jti cache, injects
//! `CapabilitySet` + `UserId` + `TenantContext` into extensions.

use std::sync::Arc;
use std::time::Duration;

use axum::{
    extract::{Request, State},
    http::{header, StatusCode},
    middleware::Next,
    response::Response,
};
use moka::future::Cache;
use nas2_application::auth::{JwtClaims, JwtVerifier};
use nas2_common::{AppError, TenantId, UserId};
use nas2_domain::{Capability, CapabilitySet};
use nas2_tenant::TenantContext;
use uuid::Uuid;

#[derive(Clone)]
pub struct JwtMiddlewareState {
    pub verifier: Arc<dyn JwtVerifier>,
    pub jti_cache: Cache<Uuid, ()>,
}

impl JwtMiddlewareState {
    pub fn new(verifier: Arc<dyn JwtVerifier>) -> Self {
        Self {
            verifier,
            // Max 100k entries, TTL = 1h default (token lifespan).
            jti_cache: Cache::builder()
                .max_capacity(100_000)
                .time_to_live(Duration::from_secs(3600))
                .build(),
        }
    }
}

/// Extract token from Bearer header (precedence) or `ax_session` cookie.
fn extract_token(req: &Request) -> Option<String> {
    if let Some(auth) = req.headers().get(header::AUTHORIZATION) {
        if let Ok(s) = auth.to_str() {
            if let Some(t) = s.strip_prefix("Bearer ") {
                return Some(t.to_owned());
            }
        }
    }
    if let Some(cookie) = req.headers().get(header::COOKIE) {
        if let Ok(s) = cookie.to_str() {
            for pair in s.split(';') {
                if let Some((k, v)) = pair.trim().split_once('=') {
                    if k == "ax_session" {
                        return Some(v.to_owned());
                    }
                }
            }
        }
    }
    None
}

pub async fn jwt_middleware(
    State(state): State<JwtMiddlewareState>,
    mut req: Request,
    next: Next,
) -> Result<Response, AppError> {
    let token = extract_token(&req)
        .ok_or_else(|| AppError::Forbidden("missing_token".to_owned()))?;
    let claims = state.verifier.verify(&token).await?;

    // jti replay check
    if state.jti_cache.get(&claims.jti).await.is_some() {
        tracing::warn!(jti = %claims.jti, "jwt replay detected");
        return Err(AppError::Forbidden("replay_detected".to_owned()));
    }
    state.jti_cache.insert(claims.jti, ()).await;

    // Build CapabilitySet from claims.caps strings
    let mut set = CapabilitySet::new();
    for c in &claims.caps {
        if let Ok(cap) = Capability::try_from_str(c) {
            set.insert(cap);
        }
        // Unknown caps silently dropped — forward-compat: clients can
        // claim capabilities the server doesn't recognise without 401.
    }

    // Tenant context expected from upstream resolve_tenant. If the
    // JWT's tenant differs from the resolved context, that's
    // suspicious. Fail closed.
    if let Some(existing) = req.extensions().get::<TenantContext>() {
        if existing.tenant_id != claims.tenant {
            tracing::warn!(
                resolver_tenant = %existing.tenant_id,
                jwt_tenant = %claims.tenant,
                "tenant mismatch between resolver and JWT"
            );
            return Err(AppError::Forbidden("tenant_mismatch".to_owned()));
        }
    }

    req.extensions_mut().insert(set);
    req.extensions_mut().insert(claims.sub);   // UserId
    req.extensions_mut().insert(claims);       // full claims if needed

    Ok(next.run(req).await)
}
```

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Cookie parsing is manual.** `tower-sessions` exists but is
     overkill for read-only JWT-in-cookie. Manual `split(';')` is
     fine for our subset.
  2. **`Cache<Uuid, ()>` moka choice.** Empty value is idiomatic for
     "have we seen this jti". TTL drives expiry; capacity drives LRU
     eviction under high churn.
  3. **Bearer precedence over cookie.** CLI tools using `-H
     "Authorization: Bearer"` must NOT also get a cookie applied.
     Our order: Bearer first, cookie fallback.
  4. **`tenant_mismatch` check** requires `resolve_tenant` to run
     BEFORE `jwt_middleware`. Order in router: outermost = resolver,
     inner = jwt. axum's Service composition handles this when both
     are layered correctly.
  5. **Forwarding unknown caps**: if JWT claims `caps: ["future.cap"]`
     and server doesn't know `future.cap`, we drop it silently. That
     keeps client+server in lockstep without 401 on every minor
     deploy delta.
