```rust
//! Paginated media list use case.

use nas2_common::{AppError, Page, TenantId};
use nas2_domain::{Capability, CapabilitySet, Media};
use crate::ports::MediaRepository;

pub struct ListMedia<R> { repo: R }

impl<R: MediaRepository> ListMedia<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        page: u32,
        per_page: u32,
        caps: &CapabilitySet,
    ) -> Result<Page<Media>, AppError> {
        if !caps.contains(Capability::MediaUpload) {
            return Err(AppError::Forbidden(Capability::MediaUpload.as_str().into()));
        }
        self.repo.list_for_tenant_paginated(tenant, page, per_page).await
    }
}
```

```rust
#[component]
fn MediaTileCard(tile: MediaTile) -> impl IntoView {
    view! {
        <li class="media-tile" data-media-id=tile.id.to_string()>
            <div class="media-thumb">
                <img src=tile.thumb_url alt=tile.alt.clone() loading="lazy"/>
            </div>
            <footer class="media-meta">
                <span class="media-mime">{tile.mime}</span>
                <span class="media-size">{human_bytes(tile.bytes_size)}</span>
            </footer>
        </li>
    }
}

fn human_bytes(n: u64) -> String {
    const KB: u64 = 1024;
    const MB: u64 = KB * 1024;
    if n >= MB { format!("{:.1} MB", n as f64 / MB as f64) }
    else if n >= KB { format!("{:.0} KB", n as f64 / KB as f64) }
    else { format!("{n} B") }
}
```

P3 · Handler `admin/media.rs::media_browser`

```rust
// caps.require(media.upload)
pub async fn media_browser(
    _: RequireCaps<MediaUpload>,
    State(state): State<AppState>,
    Extension(caps): Extension<CapabilitySet>,
    Extension(tctx): Extension<TenantContext>,
    Query(q): Query<MediaListQuery>,
) -> Result<impl IntoResponse, AppError> {
    let uc = ListMedia::new(state.media_repo.clone());
    let page = uc.execute(tctx.tenant_id, q.page, q.per_page, &caps).await?;
    let items: Vec<MediaTile> = page.items.into_iter().map(|m| {
        let thumb_url = m.variants.iter()
            .find(|v| v.kind == "thumb_320")
            .map(|v| v.url.clone())
            .unwrap_or_else(|| m.original_url.clone());
        MediaTile {
            id: m.id,
            thumb_url,
            alt: m.alt,
            mime: m.mime,
            bytes_size: m.bytes_size,
        }
    }).collect();
    let initial = Page::new(items, page.total, page.page, page.per_page);
    let ctx = build_admin_ctx(/* ... */);
    let crumbs = BreadcrumbChain::root().push(Crumb::terminal("Media"));
    let html = leptos::ssr::render_to_string(move || view! {
        <AdminLayout ctx title="Media" crumbs>
            <MediaBrowser initial/>
        </AdminLayout>
    });
    Ok(Html(html.into_owned()))
}
```

P5 · Intersection Observer for infinite scroll (small JS)

  Append to `admin/static/session.js` OR create separate `media.js`:

```javascript
(function () {
  function observe() {
    const sentinel = document.querySelector('.load-more-sentinel');
    if (!sentinel) return;
    const obs = new IntersectionObserver((entries) => {
      if (entries.some(e => e.isIntersecting)) {
        sentinel.click();  // triggers Leptos handler
      }
    });
    obs.observe(sentinel);
  }
  if (document.readyState === 'complete') observe();
  else window.addEventListener('load', observe);
})();
```
