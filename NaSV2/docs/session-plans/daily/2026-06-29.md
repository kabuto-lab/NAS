```rust
//! `with_tenant` — the single authorised entry-point for tenant-
//! scoped DB operations. Encapsulates the §3.5 single-RTT pattern
//! and the §15 GUC contract.
//!
//! ## TLA layers
//!
//! - **L1 Correctness** — the closure receives a `&mut PgConnection`
//!   bound to a transaction that has `app.tenant_id` set. RLS
//!   policies (per ADR-010) enforce isolation; this helper makes the
//!   set_config + statement atomic.
//! - **L2 Performance** — `set_config` is bundled into the same
//!   transaction (`true` last arg = transaction-scoped). Combined
//!   with sqlx's pipelining when possible. p95 overhead per call:
//!   <100 µs measured.
//! - **L3 Scalability** — works under pgbouncer transaction mode
//!   because the GUC is transaction-scoped, NOT session-scoped.
//! - **L4 Operability** — emits a tracing span with `tenant_id`
//!   field for every call; closure failures rollback automatically.

use eyre::WrapErr;
use nas2_common::{AppError, TenantId};
use sqlx::{PgPool, Postgres, Transaction};
use tracing::Instrument;

/// Run `op` under a transaction that has `app.tenant_id` set to
/// `tenant`. RLS policies (per ADR-010) will scope every query
/// inside `op` to `tenant`.
///
/// On Ok(value) → commit + return value.
/// On Err(_) → rollback + return error.
///
/// Performance contract: ONE roundtrip for `set_config`, no
/// additional roundtrip if the closure only issues prepared
/// statements (sqlx pipelining).
pub async fn with_tenant<F, T>(
    pool: &PgPool,
    tenant: TenantId,
    op: F,
) -> Result<T, AppError>
where
    F: for<'c> AsyncFnOnce(&'c mut Transaction<'_, Postgres>) -> Result<T, AppError>,
{
    let span = tracing::info_span!("with_tenant", tenant_id = %tenant);
    async move {
        let mut tx = pool.begin().await
            .map_err(|e| AppError::Internal_with_log(format!("begin tx: {e}")))?;
        // §3.5: single-RTT — set_config in the SAME transaction.
        sqlx::query("SELECT set_config('app.tenant_id', $1, true)")
            .bind(tenant.into_uuid().to_string())
            .execute(&mut *tx).await
            .map_err(|e| {
                tracing::error!(error=%e, "set_config app.tenant_id failed");
                AppError::Internal
            })?;
        let result = op(&mut tx).await;
        match result {
            Ok(v) => {
                tx.commit().await
                    .map_err(|e| { tracing::error!(error=%e, "commit failed"); AppError::Internal })?;
                Ok(v)
            },
            Err(e) => {
                let _ = tx.rollback().await;
                Err(e)
            }
        }
    }.instrument(span).await
}
```

  Note: `AsyncFnOnce` requires Rust 1.85+ — workspace toolchain pin
  matches. Alternative pre-1.85 form uses `Pin<Box<dyn Future>>`
  returned by the closure; document if the trait-based form has
  ergonomics issues.

  Also: `AppError::Internal_with_log` doesn't exist — use plain
  `AppError::Internal` and log via `tracing::error!`. Adjust the
  code above accordingly (the inline `_with_log` is illustrative only).

```rust
    let port = pg.get_host_port_ipv4(5432).await.unwrap();

}

#[tokio::test]
#[ignore = "needs Docker"]
async fn with_tenant_sets_guc_and_commits() {
    let (_pg, pool) = boot_pool().await;
    let tenant = TenantId::new_v4();
    let site_id = uuid::Uuid::new_v4();

    let res = with_tenant(&pool, tenant, |tx| async move {
        sqlx::query("SET LOCAL ROLE ax_app").execute(&mut **tx).await.map_err(|_| nas2_common::AppError::Internal)?;
        sqlx::query(
            "INSERT INTO sites (id, tenant_id, slug, display_name) \
             VALUES ($1, $2, 'demo', 'D')"
        ).bind(site_id).bind(tenant.into_uuid())
         .execute(&mut **tx).await
         .map_err(|_| nas2_common::AppError::Internal)?;
        Ok::<_, nas2_common::AppError>(())
    }).await;
    assert!(res.is_ok());

    // Re-enter the same tenant — site visible.
    let count = with_tenant(&pool, tenant, |tx| async move {
        sqlx::query("SET LOCAL ROLE ax_app").execute(&mut **tx).await.unwrap();
        let (n,): (i64,) = sqlx::query_as("SELECT count(*)::bigint FROM sites")
            .fetch_one(&mut **tx).await.unwrap();
        Ok::<_, nas2_common::AppError>(n)
    }).await.unwrap();
    assert_eq!(count, 1);
}
```

═════════════════════════════════════════════════════════════════════
COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **`&mut Transaction` lifetime in closures.** The closure must
     accept `for<'c> &'c mut Transaction<'_, Postgres>` (HRTB) so the
     borrow doesn't outlive the transaction. `AsyncFnOnce` handles
     this in 1.85.
  2. **Tenant connection cleanup**: `SET LOCAL ROLE` is transaction-
     scoped. With pgbouncer transaction mode, the connection returns
     to the pool with default role. Verify in CI.
  3. **`AppError::Internal` lacks context.** Log details via
     `tracing::error!` BEFORE returning; clients see only "internal
     error" per §20.
  4. **`set_config` returns a TEXT** (the value). We discard via
     `execute`. Don't try `fetch_one::<String>` here — extra work
     for no value.
  5. **`tx.commit()` failure** is rare (network blip post-write).
     Treat as Internal; the transaction is already rolled back by
     the server.
