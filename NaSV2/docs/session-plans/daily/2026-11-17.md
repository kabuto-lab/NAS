use nas2_common::{AppError, Page, SiteId, TenantId};
use nas2_domain::{Capability, CapabilitySet, Post, PostStatus};
use crate::ports::PostRepository;

#[derive(Debug, Clone, Copy, Default, serde::Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum PostSortOrder {
    #[default]
    UpdatedDesc,
    UpdatedAsc,
    TitleAsc,
    TitleDesc,
    PublishedDesc,
}

pub struct ListPostsAdmin<R> { repo: R }

impl<R: PostRepository> ListPostsAdmin<R> {
    pub const fn new(repo: R) -> Self { Self { repo } }

    pub async fn execute(
        &self,
        tenant: TenantId,
        site: SiteId,
        page: u32,
        per_page: u32,
        sort: PostSortOrder,
        caps: &CapabilitySet,
    ) -> Result<Page<Post>, AppError> {
        if !caps.contains(Capability::CmsPageRead) {
            return Err(AppError::Forbidden(Capability::CmsPageRead.as_str().into()));
        }
        self.repo.list_admin(tenant, site, page, per_page, sort).await
    }
}

  `crates/infrastructure/src/persistence/post.rs` — add method:

```rust
async fn list_admin(
    &self,
    tenant: TenantId,
    site: SiteId,
    page: u32,
    per_page: u32,
    sort: nas2_application::queries::PostSortOrder,
) -> Result<Page<Post>, AppError> {
    use nas2_application::queries::PostSortOrder::*;
    let per_page = per_page.clamp(1, 100);
    let page = page.max(1);
    let offset = u64::from(page - 1) * u64::from(per_page);

    let order_clause = match sort {
        UpdatedDesc   => "updated_at DESC, id",
        UpdatedAsc    => "updated_at ASC, id",
        TitleAsc      => "title ASC, id",
        TitleDesc     => "title DESC, id",
        PublishedDesc => "published_at DESC NULLS LAST, id",
    };

    let sql = format!(
        "SELECT id, site_id, tenant_id, slug, title, status, \
                published_at, blocks, custom_type, \
                COUNT(*) OVER() AS total \
         FROM posts WHERE site_id = $1 \
         ORDER BY {order_clause} \
         LIMIT $2 OFFSET $3"
    );

    with_tenant(&self.pool, tenant, |tx| async move {
        let rows: Vec<(PostRow, i64)> = sqlx::query_as::<_, (PostRow, i64)>(&sql)
            .bind(site.into_uuid())
            .bind(i64::from(per_page))
            .bind(i64::try_from(offset).map_err(|_| AppError::Validation("offset overflow".into()))?)
            .fetch_all(&mut **tx).await
            .map_err(|e| { tracing::error!(error=%e, "list_admin"); AppError::Internal })?;
        let total = rows.first().map(|(_,t)| u64::try_from(*t).unwrap_or(0)).unwrap_or(0);
        let items: Result<Vec<_>, _> = rows.into_iter().map(|(r,_)| r.into_domain()).collect();
        Ok(Page::new(items?, total, page, per_page))
    }).await
}
```

  ORDER BY clause is built from a CLOSED ENUM — no SQL injection
  risk. Format string is fine; PG plan caches by query text shape.

#[component]
pub fn PostListPager(
    page: u32,
    per_page: u32,
    total: u64,
    sort_qs: String,   // pre-encoded `&sort=...` etc.
) -> impl IntoView {
    let total_pages = if per_page == 0 { 1 } else { ((total + per_page as u64 - 1) / per_page as u64).max(1) as u32 };
    let prev_disabled = page <= 1;
    let next_disabled = page >= total_pages;
    let mk = |p: u32| format!("?page={p}&per_page={per_page}{sort_qs}");
    view! {
        <nav class="pager" aria-label="pagination">
            <a href=mk((page.saturating_sub(1)).max(1))
               class:disabled=prev_disabled aria-disabled=prev_disabled>"← Prev"</a>
            <span class="pager-pages">
                {page_window(page, total_pages).into_iter().map(|p| {
                    let cur = p == page;
                    view! {
                        <a href=mk(p) class:current=cur aria-current=cur>{p}</a>
                    }
                }).collect_view()}
            </span>
            <a href=mk((page + 1).min(total_pages))
               class:disabled=next_disabled aria-disabled=next_disabled>"Next →"</a>
            <PerPageSelector per_page page sort_qs/>
        </nav>
    }
}

fn page_window(current: u32, total: u32) -> Vec<u32> {
    // Show first 2, current ±1, last 2 — collapse middle with "…"
    // implementation as a Vec for simplicity; ellipsis rendered as
    // a `p=0` sentinel
    let mut s = std::collections::BTreeSet::new();
    s.insert(1);
    if total > 1 { s.insert(total); }
    for p in [current.saturating_sub(1), current, current + 1] {
        if p >= 1 && p <= total { s.insert(p); }
    }
    s.into_iter().collect()
}

#[component]
fn PerPageSelector(per_page: u32, page: u32, sort_qs: String) -> impl IntoView {
    let opts = [10, 25, 50, 100];
    view! {
        <form method="get" class="per-page-form">
            <input type="hidden" name="page" value=page.to_string()/>
            // Carry sort param too — parse sort_qs into hidden inputs
            // (omitted in skeleton; implement)
            <label>"Per page: "
                <select name="per_page" onchange="this.form.submit()">
                    {opts.into_iter().map(|n| view! {
                        <option value=n.to_string() selected=(n == per_page)>{n}</option>
                    }).collect_view()}
                </select>
            </label>
        </form>
    }
}

#[component]
pub fn PostListSortHeaders(current: nas2_application::queries::PostSortOrder, base_qs: String) -> impl IntoView {
    use nas2_application::queries::PostSortOrder::*;
    let mk = |s: &str, label: &str, asc: PostSortOrder, desc: PostSortOrder| {
        let (next, arrow) = match (current, asc, desc) {
            (cur, a, _) if cur == a => (desc.clone(), "↓"),
            (cur, _, d) if cur == d => (a.clone(), "↑"),
            _ => (a.clone(), ""),
        };
        let qs = format!("{base_qs}&sort={}", serde_qs_value(&next));
        view! {
            <th><a href=qs>{label}<span class="sort-arrow">{arrow}</span></a></th>
        }
    };
    view! {
        <tr>
            {mk("title", "Title", TitleAsc, TitleDesc)}
            <th>"Status"</th>
            <th>"Author"</th>
            {mk("updated", "Updated", UpdatedAsc, UpdatedDesc)}
            <th class="actions-col">"Actions"</th>
        </tr>
    }
}

fn serde_qs_value(s: &nas2_application::queries::PostSortOrder) -> &'static str {
    use nas2_application::queries::PostSortOrder::*;
    match s { UpdatedDesc => "updated_desc", UpdatedAsc => "updated_asc",
              TitleAsc => "title_asc", TitleDesc => "title_desc",
              PublishedDesc => "published_desc" }
}

COMMON PITFALLS / PERF / SEC
═════════════════════════════════════════════════════════════════════

  1. **ORDER BY interpolated** — closed enum → no SQL injection.
     Verify pattern: never accept raw user string for ORDER BY.
  2. **`NULLS LAST` for `published_at DESC`** — drafts have NULL;
     placing them last keeps the published-first ordering intuitive.
  3. **Per-page > 100** — caller would bypass via crafted URL;
     `clamp(1, 100)` enforces. Tests cover.
  4. **`<a href>` for pagination + sort** = GET-only, idempotent,
     bookmarkable, no JS. Pure progressive-enhancement-ready.
  5. **Sort arrow accessibility** — current sort marked with
     `aria-sort="ascending|descending"` would be ideal; deferred to
     a11y polish.
