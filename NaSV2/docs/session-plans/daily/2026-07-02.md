P1 · `migrations/0008_outbox_search_reindex.sql` (NEW)

```sql
-- 0008_outbox_search_reindex.sql · transactional outbox for
-- ENTITY §3.8 (queue tier). Writes by repos are committed in the
-- same transaction as the business write; M11 search worker drains
-- + emits to pgmq.
--
-- ADR-010 RLS shape applies (tenant_id + ax_app grant + force RLS).

CREATE TABLE IF NOT EXISTS outbox_search_reindex (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id    UUID NOT NULL,
    aggregate    TEXT NOT NULL
                 CHECK (aggregate IN ('post','term','media','comment')),
    aggregate_id UUID NOT NULL,
    op           TEXT NOT NULL CHECK (op IN ('insert','update','delete')),
    enqueued_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS outbox_search_reindex_pending_idx
    ON outbox_search_reindex (enqueued_at);

CREATE INDEX IF NOT EXISTS outbox_search_reindex_tenant_idx
    ON outbox_search_reindex (tenant_id);

ALTER TABLE outbox_search_reindex ENABLE ROW LEVEL SECURITY;
ALTER TABLE outbox_search_reindex FORCE ROW LEVEL SECURITY;
CREATE POLICY outbox_search_reindex_tenant_isolation ON outbox_search_reindex
    USING       (tenant_id = current_setting('app.tenant_id')::uuid)
    WITH CHECK  (tenant_id = current_setting('app.tenant_id')::uuid);
GRANT SELECT, INSERT, DELETE ON outbox_search_reindex TO ax_app;
```

async fn insert(&self, post: &Post) -> Result<(), AppError> {
    let blocks = serde_json::to_value(&post.blocks)
        .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
    let status = match post.status {
        PostStatus::Draft     => "draft",
        PostStatus::Scheduled => "scheduled",
        PostStatus::Published => "published",
        PostStatus::Archived  => "archived",
    };
    let post_id    = post.id.into_uuid();
    let site_id    = post.site_id.into_uuid();
    let tenant_id  = post.tenant_id.into_uuid();
    let slug       = post.slug.as_str().to_owned();
    let title      = post.title.clone();
    let pub_at     = post.published_at;
    let cust       = post.custom_type.as_ref().map(|c| c.0.clone());

    with_tenant(&self.pool, post.tenant_id, |tx| async move {
        sqlx::query(
            "INSERT INTO posts (id, site_id, tenant_id, slug, title, status, \
                                 published_at, blocks, custom_type) \
             VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)"
        )
            .bind(post_id).bind(site_id).bind(tenant_id)
            .bind(&slug).bind(&title).bind(status)
            .bind(pub_at).bind(&blocks).bind(cust)
            .execute(&mut **tx).await
            .map_err(|e| classify_pg_err(e, "post.insert"))?;

        // Search-reindex outbox row (same tx — atomic).
        sqlx::query(
            "INSERT INTO outbox_search_reindex (tenant_id, aggregate, aggregate_id, op) \
             VALUES ($1, 'post', $2, 'insert')"
        ).bind(tenant_id).bind(post_id)
         .execute(&mut **tx).await
         .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;

        Ok::<_, AppError>(())
    }).await
}

async fn update(&self, post: &Post) -> Result<(), AppError> {
    let blocks = serde_json::to_value(&post.blocks)
        .map_err(|_| AppError::Internal)?;
    let status = match post.status {
        PostStatus::Draft     => "draft",
        PostStatus::Scheduled => "scheduled",
        PostStatus::Published => "published",
        PostStatus::Archived  => "archived",
    };
    let post_id = post.id.into_uuid();
    let title   = post.title.clone();
    let pub_at  = post.published_at;

    with_tenant(&self.pool, post.tenant_id, |tx| async move {
        let result = sqlx::query(
            "UPDATE posts SET title = $1, status = $2, published_at = $3, \
                              blocks = $4 \
             WHERE id = $5"
        )
            .bind(&title).bind(status).bind(pub_at).bind(&blocks)
            .bind(post_id)
            .execute(&mut **tx).await
            .map_err(|e| classify_pg_err(e, "post.update"))?;
        if result.rows_affected() == 0 {
            return Err(AppError::NotFound(format!("post {post_id}")));
        }
        sqlx::query(
            "INSERT INTO outbox_search_reindex (tenant_id, aggregate, aggregate_id, op) \
             VALUES ($1, 'post', $2, 'update')"
        ).bind(post.tenant_id.into_uuid()).bind(post_id)
         .execute(&mut **tx).await
         .map_err(|e| { tracing::error!(error=%e); AppError::Internal })?;
        Ok::<_, AppError>(())
    }).await
}

// Top-level helper used by both insert + update.
fn classify_pg_err(e: sqlx::Error, context: &str) -> AppError {
    if let sqlx::Error::Database(ref db) = e {
        if db.code().as_deref() == Some("23505") {
            // unique_violation
            return AppError::Conflict(db.message().to_owned());
        }
    }
    tracing::error!(error=%e, "{context} db error");
    AppError::Internal
}

             WHERE aggregate = 'post' AND op = 'insert'"

        Ok::<_, nas2_common::AppError>(n)
    }).await.unwrap();
    assert_eq!(count, 1, "insert outbox row must exist");
}

COMMON PITFALLS
═════════════════════════════════════════════════════════════════════

  1. **Transactional outbox**: writes-to-outbox MUST be in the same
     TX as the business write. M11's worker reads + deletes by id
     after successful pgmq enqueue. Without atomicity, you have
     dual-write inconsistency.
  2. **`gen_random_uuid()`** requires the pgcrypto extension. Ship
     `CREATE EXTENSION IF NOT EXISTS pgcrypto` in 0008 OR use UUIDv7
     generated at insert time (cleaner). Choice: pgcrypto in migration.
  3. **`rows_affected() == 0`** distinguishes NotFound from real
     update conflict. Without the check, callers can't tell.
  4. **`classify_pg_err` doesn't cover FK violations** (23503). M2
     doesn't have user-input that would hit them; M3 onwards might
     (uploads with bad media refs).
  5. **`Block::Paragraph { html }` deserialisation**: the seed test
     uses Paragraph because it has no FK; Image variant would need
     a real media row.
