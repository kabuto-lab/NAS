/// Catalog of insertable block variants — kept here (NOT in domain)

pub const VARIANTS: &[VariantMeta] = &[

#[component]
pub fn InsertButton(at_index: usize) -> impl IntoView {
    let palette_open = RwSignal::new(false);
    view! {
        <div class="insert-gap">
            <button class="insert-toggle" type="button"
                    on:click=move |_| palette_open.update(|v| *v = !*v)
                    aria-label=format!("Insert block at position {at_index}")>
                "+"
            </button>
            <Show when=move || palette_open.get() fallback=|| ()>
                <BlockPalette at_index on_close=move || palette_open.set(false)/>
            </Show>
        </div>
    }
}

#[component]
pub fn BlockPalette(
    at_index: usize,
    #[prop(into)] on_close: Callback<()>,
) -> impl IntoView {
    let state = expect_context::<EditorState>();
    let groups: std::collections::BTreeMap<&str, Vec<&VariantMeta>> = {
        let mut m = std::collections::BTreeMap::new();
        for v in VARIANTS { m.entry(v.group).or_insert_with(Vec::new).push(v); }
        m
    };
    let insert = move |variant: &'static VariantMeta| {
        let new_block = (variant.default_block)();
        state.blocks.update(|blocks| {
            let i = at_index.min(blocks.len());
            blocks.insert(i, new_block);
        });
        // Move focus into the new block (set editing_index)
        state.editing_index.set(Some(at_index));
        on_close.call(());
    };
    view! {
        <div class="palette-modal" role="dialog" aria-label="block palette">
            <div class="palette-backdrop" on:click=move |_| on_close.call(())></div>
            <div class="palette-content">
                <header>
                    <h3>"Insert block"</h3>
                    <button class="palette-close" type="button"
                            on:click=move |_| on_close.call(())>"×"</button>
                </header>
                {groups.into_iter().map(|(g, items)| view! {
                    <section>
                        <h4>{g}</h4>
                        <ul class="palette-grid">
                            {items.into_iter().map(|v| {
                                let kind = v.kind;
                                let label = v.label;
                                view! {
                                    <li>
                                        <button type="button" class="palette-item"
                                                on:click=move |_| insert(v)>
                                            <span class="palette-icon">{first_letter(kind)}</span>
                                            <span class="palette-label">{label}</span>
                                        </button>
                                    </li>
                                }
                            }).collect_view()}
                        </ul>
                    </section>
                }).collect_view()}
            </div>
        </div>
    }
}

fn first_letter(s: &str) -> char {
    s.chars().next().unwrap_or('?').to_ascii_uppercase()
}

PERFORMANCE / SECURITY / PITFALLS
═════════════════════════════════════════════════════════════════════

  - Palette modal: zero-cost when closed (`<Show when>` only mounts
    when open).
  - Default-constructed blocks have empty strings; `Block::validate()`
    (M4 W3) may reject pure-empty values at save time — that's
    intentional (forces operator to fill before save).
  - `Callback<()>` for on_close: Leptos pattern, no closure-clone overhead.
  - VARIANTS table is a `&'static [VariantMeta]` — zero alloc;
    `default_block: fn() -> Block` is a function pointer, no
    closure capture issues.
  - Palette grid uses CSS auto-fill — responsive without JS.
